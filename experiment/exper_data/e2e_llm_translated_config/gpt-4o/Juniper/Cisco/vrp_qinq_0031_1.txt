hostname CE2

 encapsulation dot1Q 20
 ip address 10.3.1.2 255.255.255.0
!

router bgp [your_local_as]
 neighbor 10.2.1.1 remote-as 100
 neighbor 10.2.1.1 activate
!

!

 class class-default
  set continue
!

!
