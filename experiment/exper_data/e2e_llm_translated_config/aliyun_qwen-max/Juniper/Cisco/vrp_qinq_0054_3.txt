hostname CE1

interface GigabitEthernet1/0/1
 switchport mode trunk
 switchport trunk allowed vlan 1
!
interface GigabitEthernet1/0/1.1
 encapsulation dot1q 1
 ip address 192.168.1.2 255.255.255.0
!
interface GigabitEthernet1/0/2
 switchport mode trunk
 switchport trunk allowed vlan 1
!
interface GigabitEthernet1/0/2.1
 encapsulation dot1q 1
 ip address 172.16.1.2 255.255.255.0
!

router bgp 65410
 bgp router-id 192.168.1.2
 neighbor 192.168.1.1 remote-as 100
 neighbor 172.16.1.1 remote-as 100
!
ip prefix-list IMPORT-DIRECT seq 5 permit 0.0.0.0/0 le 32
route-map IMPORT-DIRECT permit 10
 match ip address prefix-list IMPORT-DIRECT
!
neighbor 192.168.1.1 route-map IMPORT-DIRECT in
neighbor 172.16.1.1 route-map IMPORT-DIRECT in
!
exit
