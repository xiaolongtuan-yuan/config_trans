hostname CE2

vlan 10

ethernet cfm

interface Vlan10
 ip address 10.1.1.2 255.255.255.0

interface GigabitEthernet1/0/0
 switchport mode trunk
 switchport trunk allowed vlan 10
 no shutdown

interface GigabitEthernet2/0/0
 switchport mode trunk
 switchport trunk allowed vlan 10
 no shutdown

ethernet cfm md md1
 mep domain ma1
  mep domain ma1 mep-id 2 interface GigabitEthernet1/0/0 direction outward
  mep domain ma1 mep-id 2 ccm-send
  mep domain ma1 mep-id 1 remote-mep
  mep domain ma1 mep-id 1 ccm-receive

ethernet cfm md md2
 mep domain ma2
  mep domain ma2 mep-id 4 interface GigabitEthernet2/0/0 direction outward
  mep domain ma2 mep-id 4 ccm-send
  mep domain ma2 mep-id 3 remote-mep
  mep domain ma2 mep-id 3 ccm-receive

end
