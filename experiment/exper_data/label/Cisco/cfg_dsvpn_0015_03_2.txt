hostname Spoke2

interface GigabitEthernet1/0/0
 ip address 10.1.3.10 255.255.255.0
 tunnel source GigabitEthernet2/0/0
 tunnel mode gre multipoint
 tunnel key 1
 tunnel protection ipsec profile IPSEC_PROFILE_NAME

interface GigabitEthernet2/0/0
 ip address 192.168.2.1 255.255.255.0

interface Tunnel0
 ip address 172.16.1.3 255.255.255.0
 tunnel source GigabitEthernet1/0/0
 tunnel destination 172.16.1.1
 tunnel mode gre multipoint
 tunnel key 1
 tunnel protection ipsec profile IPSEC_PROFILE_NAME

router bgp 300
 bgp router-id 172.16.1.3
 neighbor 172.16.1.1 remote-as 100
 neighbor 172.16.1.2 remote-as 200
 neighbor 172.16.1.1 activate
 neighbor 172.16.1.2 activate
 address-family ipv4 unicast
  neighbor 172.16.1.1 activate
  neighbor 172.16.1.2 activate
  redistribute ospf 1
 exit-address-family

router ospf 1
 router-id 172.16.1.3
 network 192.168.2.0 0.0.0.255 area 0
 redistribute bgp 300 subnets

router ospf 2
 router-id 10.1.3.10
 network 10.1.3.0 0.0.0.255 area 1
