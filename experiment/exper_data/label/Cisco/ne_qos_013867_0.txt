#
ip multicast-routing
#
class-map match-any huawei-qos
 match access-group name huawei-qos-acl
!
policy-map huawei-qos
 class huawei-qos
  bandwidth remaining percent 100
!
interface GigabitEthernet1/0/0
 no shutdown
 ip igmp version 3
 ip igmp explicit-tracking
 ip igmp snooping vlan 1
!
interface GigabitEthernet1/0/0.100
 ip igmp version 3
 encapsulation dot1Q 100
 ip igmp snooping vlan 100
 service-policy input huawei-qos
!
interface GigabitEthernet2/0/0
 no shutdown
 ip address 10.0.0.3 255.255.255.0
 ip pim sparse-mode
!
ip multicast-routing distributed
!
ip access-list standard huawei-qos-acl
 permit 239.0.0.0 0.0.0.255
!
ip multicast-routing profile huawei
 ip multicast-routing reserved-bandwidth 1024
!
ip multicast-routing profile cctv
 ip multicast-routing detect-interval 15
 ip multicast-routing threshold 15
!
ip multicast-routing profile jstv
 ip multicast-routing detect-interval 15
 ip multicast-routing threshold 20
!
return
