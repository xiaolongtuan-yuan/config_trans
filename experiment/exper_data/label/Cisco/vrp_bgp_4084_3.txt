sysname DeviceD
!
interface GigabitEthernet1/0/0
 no shutdown
 ip address 10.3.1.2 255.255.255.0
!
interface GigabitEthernet2/0/0
 no shutdown
 ip address 10.4.1.2 255.255.255.0
!
interface LoopBack1
 ip address 4.4.4.4 255.255.255.255
!
router bgp 200
 bgp router-id 4.4.4.4
 neighbor 10.3.1.1 remote-as 200
 neighbor 10.3.1.1 fall-over bfd min-tx-interval 100 min-rx-interval 100 multiplier 4
 neighbor 10.3.1.1 fall-over bfd
 neighbor 10.4.1.1 remote-as 200
 neighbor 10.4.1.1 fall-over bfd min-tx-interval 100 min-rx-interval 100 multiplier 4
 neighbor 10.4.1.1 fall-over bfd
 !
 address-family ipv4 unicast
  synchronization
  network 4.4.4.4 mask 255.255.255.255
  neighbor 10.3.1.1 activate
  neighbor 10.4.1.1 activate
exit-address-family
!
end
