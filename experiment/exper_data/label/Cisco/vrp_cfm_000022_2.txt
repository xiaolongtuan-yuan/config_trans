sysname DeviceC
vlan batch 2 to 4
cfm enable
interface GigabitEthernet1/0/0
 portswitch
 port trunk allow-pass vlan 2 to 4
interface GigabitEthernet1/0/1
 portswitch
 port trunk allow-pass vlan 2 to 4
cfm md md1 level 6
 ma ma1
  map vlan 2
 ma ma2
  map vlan 3
cfm md md2 level 4
 ma ma3
  map vlan 4
  mep mep-id 2 interface gigabitethernet 1/0/0 outward
  mep ccm-send mep-id 2 enable
  remote-mep mep-id 1
  remote-mep ccm-receive mep-id 1 enable
  remote-mep mep-id 3
  remote-mep ccm-receive mep-id 3 enable
return


hostname DeviceC
vlan batch 2-4
ethernet cfm
interface GigabitEthernet1/0/0
 switchport mode trunk
 switchport trunk allowed vlan 2-4
interface GigabitEthernet1/0/1
 switchport mode trunk
 switchport trunk allowed vlan 2-4
ethernet cfm md md1 level 6
 mep ma1
  mep domain vlan 2
 mep ma2
  mep domain vlan 3
ethernet cfm md md2 level 4
 mep ma3
  mep domain vlan 4
  mep mep-id 2 interface GigabitEthernet1/0/0 direction egress
  mep mep-id 2 ccm-send interval 1s
  rmep mep-id 1
   rmep mep-id 1 ccm-receive
  rmep mep-id 3
   rmep mep-id 3 ccm-receive
end
