set system host-name PE1

# Spanning Tree Configuration
set protocols rstp interface all mode rapid-pvst
set protocols rstp interface all path-cost method long
set protocols rstp vlan-id-list 1-10 priority 4   # Equivalent to "root primary" in Cisco
set protocols rstp vlan-id-list 11-20 priority 8  # Equivalent to "root secondary" in Cisco

# Interface GigabitEthernet1/0/1 Configuration
set interfaces ge-0/0/1 unit 0 family ethernet-switching port-mode trunk
set interfaces ge-0/0/1 unit 0 family ethernet-switching vlan members dot1q-1-20
set interfaces ge-0/0/1 disable false
set ethernet-switching-options storm-control interface ge-0/0/1 recovery-time 300
set protocols rstp interface ge-0/0/1 edge  # Equivalent to "spanning-tree portfast"
set protocols rstp interface ge-0/0/1 root-protection

# Interface GigabitEthernet1/0/1.1 Configuration
set interfaces ge-0/0/1 unit 1 encapsulation vlan-bridge
set interfaces ge-0/0/1 unit 1 vlan-id 10  # Setting it to bridge-domain

# Interface GigabitEthernet1/0/2 Configuration
set interfaces ge-0/0/2 unit 0 family ethernet-switching port-mode trunk
set interfaces ge-0/0/2 unit 0 family ethernet-switching vlan members dot1q-1-20
set interfaces ge-0/0/2 disable false
set protocols rstp interface ge-0/0/2 edge  # Equivalent to "spanning-tree portfast"

# Interface GigabitEthernet1/0/2.1 Configuration
set interfaces ge-0/0/2 unit 1 encapsulation vlan-bridge
set interfaces ge-0/0/2 unit 1 vlan-id 10  # Setting it to bridge-domain
