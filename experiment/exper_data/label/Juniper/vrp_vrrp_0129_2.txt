# 设置主机名
set system host-name PE1

# 启用MPLS并配置LDP协议
set protocols mpls
set protocols ldp interface all
set protocols ldp transport-address interface lo0.0

# 配置Loopback接口
set interfaces lo0 unit 0 family inet address 3.3.3.3/32

# 配置GigabitEthernet1/0/2接口
set interfaces ge-0/0/2 unit 0 family inet address 172.16.34.3/24
set protocols mpls interface ge-0/0/2

# 配置GigabitEthernet1/0/1接口
set interfaces ge-0/0/1 unit 0 family ethernet-switching

# 配置GigabitEthernet1/0/1.1子接口
set interfaces ge-0/0/1 unit 1 vlan-id 10
set interfaces ge-0/0/1 unit 1 family ethernet-switching

# 配置GigabitEthernet1/0/3接口
set interfaces ge-0/0/3 unit 0 family inet address 172.16.35.3/24
set protocols mpls interface ge-0/0/3

# 配置OSPF协议
set protocols ospf area 0.0.0.0 interface lo0.0
set protocols ospf area 0.0.0.0 interface ge-0/0/2
set protocols ospf area 0.0.0.0 interface ge-0/0/3

# 确保接口都处于启用状态
set interfaces ge-0/0/2 enable
set interfaces ge-0/0/3 enable

# 将XConnect配置对应到VPLS实例（基于假设，实际配置可能需要自定义）
set routing-instances ldp1 instance-type vpls
set routing-instances ldp1 interface ge-0/0/1.1
