set system host-name CGNA

set interfaces ge-0/0/1 unit 0 family inet address 11.11.11.2/24
set security zones security-zone trust interfaces ge-0/0/0.0
set security zones security-zone untrust interfaces ge-0/0/1.0

set security nat source pool pool1 address 11.11.11.1 to 11.11.11.254

set security nat source rule-set from-inside-to-outside from zone trust
set security nat source rule-set from-inside-to-outside to zone untrust

set security nat source rule-set from-inside-to-outside rule nat-rule match source-address 192.168.10.0/24

set security nat source rule-set from-inside-to-outside rule nat-rule then source-nat pool pool1
