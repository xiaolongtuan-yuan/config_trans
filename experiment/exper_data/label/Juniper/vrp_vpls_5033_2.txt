# 设置hostname
set system host-name UPE1

# 配置MPLS LDP router-id
set protocols mpls interface lo0.0
set protocols ldp interface lo0.0
set protocols ldp router-id 4.4.4.9

# 配置MPLS
set protocols mpls

# 配置接口GigabitEthernet1/0/0
set interfaces ge-0/0/0 unit 0 family inet address 10.10.3.2/24
set interfaces ge-0/0/0 unit 0 family inet mpls

# 配置接口GigabitEthernet2/0/0
set interfaces ge-0/0/1 unit 0 family ethernet-switching

# 配置子接口GigabitEthernet2/0/0.1
set interfaces ge-0/0/1 unit 0 vlan-id 1
set interfaces ge-0/0/1 unit 0 encapsulation vlan-ccc
set interfaces ge-0/0/1 unit 0 protocol mpls
set protocols l2circuit neighbor 1.1.1.9 interface ge-0/0/1.0 virtual-circuit-id 100

# 配置Loopback接口
set interfaces lo0 unit 0 family inet address 4.4.4.9/32

# 配置OSPF
set protocols ospf area 0 interface lo0.0
set protocols ospf area 0 interface ge-0/0/0.0
