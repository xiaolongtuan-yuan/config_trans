# 设置主机名
set system host-name CE1

# 接口配置
set interfaces ge-1/0/0 unit 0 family inet address 10.1.1.2/24
set interfaces lo0 unit 1 family inet address 11.11.11.11/32

# BGP配置
set protocols bgp group external-peers type external
set protocols bgp group external-peers peer-as 100
set protocols bgp group external-peers neighbor 10.1.1.1
set protocols bgp group external-peers family inet unicast
set protocols bgp group external-peers export announce-lo0
set protocols bgp local-as 65410

# 设置策略以宣告Loopback地址
set policy-options policy-statement announce-lo0 term 1 from route-filter 11.11.11.11/32 exact
set policy-options policy-statement announce-lo0 term 1 then accept
