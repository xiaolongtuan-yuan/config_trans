set system host-name PE1

# PTP全局配置（Batfish兼容语法）
!set protocols ptp mode boundary
!set protocols ptp clock-mode boundary
!set protocols ptp domain 1
!set protocols ptp priority1 128
!set protocols ptp priority2 128
!set protocols ptp interface ge-1/0/0.0
!set protocols ptp interface ge-2/0/0.0

# 时钟源配置（Batfish兼容语法）
!set protocols ptp clock-source 1 interface-type 1pps
!set protocols ptp clock-source 1 priority 2

# 接口配置（Batfish严格模式要求）
set interfaces ge-1/0/0 unit 0 family inet address 172.16.0.1/24
!set interfaces ge-1/0/0 unit 0 family ptp
set interfaces ge-1/0/0 unit 0 disable

set interfaces ge-2/0/0 unit 0 family inet address 10.0.0.1/24
!set interfaces ge-2/0/0 unit 0 family ptp
set interfaces ge-2/0/0 unit 0 disable

# OSPF配置（保持不变）
set protocols ospf area 0.0.0.0 interface ge-1/0/0.0
set protocols ospf area 0.0.0.0 interface ge-2/0/0.0
