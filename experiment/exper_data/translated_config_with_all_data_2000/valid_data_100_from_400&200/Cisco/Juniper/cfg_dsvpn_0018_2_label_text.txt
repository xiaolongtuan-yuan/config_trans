set system host-name Spoke1

set interfaces ge-0/0/0 unit 0 family inet address 10.1.2.10/24
set interfaces lo0 unit 0 family inet address 192.168.1.1/32

set interfaces gr-0/0/0 unit 0 family inet address 172.16.1.2/24
set interfaces gr-0/0/0 unit 0 tunnel source 10.1.2.10
set interfaces gr-0/0/0 unit 0 tunnel destination 172.16.1.1
set interfaces gr-0/0/0 unit 0 encapsulation gre

set security ipsec vpn GRE_OVER_IPSEC bind-interface gr-0/0/0
set security ipsec vpn GRE_OVER_IPSEC ike ipsec-policy IPSEC_PROFILE

set protocols ospf area 0.0.0.0 interface gr-0/0/0.0
set protocols ospf area 0.0.0.0 interface lo0.0

set protocols ospf area 0.0.0.1 interface ge-0/0/0.0
