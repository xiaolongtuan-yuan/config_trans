set protocols ldp interface ge-1/0/2
set protocols ldp interface lo0.1

set interfaces ge-1/0/1 flexible-vlan-tagging
set interfaces ge-1/0/1 unit 0 family ethernet-switching interface-mode trunk
set interfaces ge-1/0/1 unit 1 vlan-id 10
set interfaces ge-1/0/1 unit 1 encapsulation vlan-ccc
