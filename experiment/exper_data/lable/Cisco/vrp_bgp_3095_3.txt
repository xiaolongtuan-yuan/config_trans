hostname RR1

interface GigabitEthernet3/0/1
 no shutdown
 ip address 172.16.4.1 255.255.255.0

interface GigabitEthernet3/0/2
 no shutdown
 ip address 172.16.2.2 255.255.255.0

interface GigabitEthernet1/0/1
 no shutdown
 ip address 172.16.6.1 255.255.255.0

router bgp 65009
 bgp router-id 4.4.4.4
 neighbor 172.16.2.1 remote-as 65009
 neighbor 172.16.4.2 remote-as 65009
 neighbor 172.16.6.2 remote-as 65009

 address-family ipv4 unicast
  synchronization
  route-reflector-client
  neighbor 172.16.2.1 send-community
  neighbor 172.16.2.1 route-policy policy1 out
  neighbor 172.16.4.2 send-community
  neighbor 172.16.4.2 route-policy policy1 out
  neighbor 172.16.6.2 send-community
  neighbor 172.16.6.2 route-policy policy1 out
 exit-address-family

route-map policy1 permit 10
 set local-preference 130

end
