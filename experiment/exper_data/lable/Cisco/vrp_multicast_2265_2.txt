hostname DeviceC
!
ip multicast-routing
!
interface GigabitEthernet1/0/0
 no shutdown
 ip address 10.1.2.2 255.255.255.0
 ip pim sparse-mode
!
interface GigabitEthernet3/0/0
 no shutdown
 ip address 10.1.3.2 255.255.255.0
 ip pim sparse-mode
 ip igmp enable
!
interface Loopback1
 ip address 3.3.3.3 255.255.255.255
 ip pim sparse-mode
!
router bgp 65002
 neighbor 10.1.2.1 remote-as 65001
 neighbor 10.1.3.1 remote-as 65001
 !
 address-family ipv4 unicast
  no synchronization
  network 3.3.3.3 mask 255.255.255.255
  network 10.1.3.0 mask 255.255.255.0
  neighbor 10.1.2.1 activate
  neighbor 10.1.2.1 route-policy policy_c2a export
  neighbor 10.1.3.1 activate
 exit-address-family
 !
 address-family ipv4 multicast
  no synchronization
  network 3.3.3.3 mask 255.255.255.255
  network 10.1.3.0 mask 255.255.255.0
  neighbor 10.1.2.1 activate
  neighbor 10.1.3.1 activate
 exit-address-family
!
router ospf 1
 area 0
  network 3.3.3.3 0.0.0.0
  network 10.1.3.0 0.0.0.255
!
route-map policy_c2a permit 10
 match ip address prefix-list p_c2a
 set metric 160
!
route-map policy_c2a permit 20
!
ip prefix-list p_c2a seq 10 permit 10.1.3.0/24
!
end
