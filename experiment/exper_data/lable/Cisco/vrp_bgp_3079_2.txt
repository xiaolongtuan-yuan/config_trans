hostname DeviceC

interface GigabitEthernet1/0/0
no shutdown
ip address 10.10.1.2 255.255.255.0

interface GigabitEthernet2/0/0
no shutdown
ip address 10.10.2.2 255.255.255.0

interface GigabitEthernet3/0/0
no shutdown
ip address 10.10.3.2 255.255.255.0

router bgp 300
bgp router-id 3.3.3.3
neighbor 10.10.1.1 remote-as 100
neighbor 10.10.2.1 remote-as 200
neighbor 10.10.3.1 remote-as 400

address-family ipv4
no synchronization
neighbor 10.10.1.1 activate
neighbor 10.10.2.1 activate
neighbor 10.10.3.1 activate
neighbor 10.10.1.1 route-map dampen-policy in
neighbor 10.10.2.1 route-map dampen-policy in
neighbor 10.10.3.1 route-map dampen-policy in

route-map dampen-policy permit 10
match ip address prefix-a
set dampening 10 1000 2000 5000

route-map dampen-policy permit 20
match ip address prefix-b
set dampening 10 800 3000 10000

access-list prefix-a permit ip 8.0.0.0 0.255.255.255

access-list prefix-b permit ip 9.1.1.0 0.0.0.255

end
