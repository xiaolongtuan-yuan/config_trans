hostname PE2
!
ipv6 unicast-routing
!
interface GigabitEthernet1/0/0
 no shutdown
 ipv6 enable
 ipv6 address 2001:DB8:20::2/96
 ipv6 ospf 1 area 0
!
interface GigabitEthernet2/0/0
 no shutdown
 ip address 192.168.2.2 255.255.255.0
!
interface Loopback1
 ipv6 enable
 ipv6 address 2001:DB8:2::2/128
 ipv6 ospf 1 area 0
!
router ospf 1
 router-id 2.2.2.2
!
router bgp 100
 bgp router-id 2.2.2.2
 neighbor 192.168.2.1 remote-as 300
 neighbor 2001:DB8:1::1 remote-as 100
 neighbor 2001:DB8:1::1 update-source Loopback1
 !
 address-family ipv4
  network 192.168.2.0 mask 255.255.255.0
  no synchronization
  neighbor 192.168.2.1 activate
  neighbor 2001:DB8:1::1 activate
  neighbor 2001:DB8:1::1 send-label
 exit-address-family
!
mpls label protocol ldp
!
mpls ldp router-id Loopback1 force
!
mpls ldp discovery transport-address Loopback1
!
mpls ip
!
end
