set system host-name CGNA

set security nat static rule-set NAT_STATIC_RULES from-zone trust
set security nat static rule-set NAT_STATIC_RULES rule STATIC_RULE_1 match source-address 192.168.10.128/25
set security nat static rule-set NAT_STATIC_RULES rule STATIC_RULE_1 then static-nat prefix 11.11.11.101/25

set security policies from-zone trust to-zone untrust policy PERMIT_ALL match source-address 192.168.10.0/24
set security policies from-zone trust to-zone untrust policy PERMIT_ALL match destination-address 11.11.11.101/25
set security policies from-zone trust to-zone untrust policy PERMIT_ALL match application any
set security policies from-zone trust to-zone untrust policy PERMIT_ALL then permit

set interfaces ge-0/0/0 unit 0 family inet address 192.168.10.1/24
set interfaces ge-0/0/0 unit 0 disable
set interfaces ge-0/0/0 unit 0 input-traffic-filter policy P1

set firewall family inet filter ACL_3001 term 1 from source-address 192.168.10.0/24
set firewall family inet filter ACL_3001 term 1 then accept

set firewall family inet filter TRAFFIC_CLASSIFIER term 1 from firewall-filter ACL_3001
set firewall family inet filter TRAFFIC_CLASSIFIER term 1 then count COUNT_MATCH
set firewall family inet filter TRAFFIC_CLASSIFIER term 1 then accept

set security nat source rule-set NAT_SRC_RULES from-zone trust
set security nat source rule-set NAT_SRC_RULES to-zone untrust
set security nat source rule-set NAT_SRC_RULES rule SRC_RULE_1 match source-address 192.168.10.0/24
set security nat source rule-set NAT_SRC_RULES rule SRC_RULE_1 then source-nat interface

set security zones security-zone trust interfaces ge-0/0/0.0 host-inbound-traffic system-services all
set security zones security-zone trust interfaces ge-0/0/0.0 host-inbound-traffic protocols all
set security zones security-zone untrust interfaces ge-0/0/0.0 host-inbound-traffic system-services all
