set system host-name Device2
set bridg-domains 10 vlans vxlan-vni 5010
set bridg-domains 20 vlans vxlan-vni 5020

set interfaces vbfd10 unit 0 family inet address 192.168.10.10/24
set interfaces vbfd20 unit 0 family inet address 192.168.20.10/24

set interfaces ge-0/0/1 unit 0 family inet address 192.168.1.2/24
set interfaces ge-0/0/2 unit 0 family inet address 192.168.2.1/24

set interfaces lo0 unit 0 family inet address 3.3.3.3/32
set forwarding-options router-id 3.3.3.3

set family vni 5010 multicast-routing
set family vni 5010 interface vbfd10
set family vni 5010 host-records host 225.0.0.1 record 5010-1

set family vni 5020 multicast-routing
set family vni 5020 interface vbfd20
set family vni 5020 host-records host 225.0.0.2 record 5020-1

set routing-options router-id 3.3.3.3
set routing-options static route 0.0.0.0/0 next-hop 192.168.1.1
set routing-options static route 0.0.0.0/0 next-hop 192.168.2.1

set protocols ospf area 0 interface ge-0/0/1.0
set protocols ospf area 0 interface ge-0/0/2.0
set protocols ospf area 0 network 3.3.3.3/32
set protocols ospf area 0 network 192.168.1.0/24
set protocols ospf area 0 network 192.168.2.0/24
