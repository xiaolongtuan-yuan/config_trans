# 设置设备主机名
set system host-name DeviceC

# 配置ISIS
set protocols isis level 2 wide-metrics-only
set protocols isis level 2-only
set protocols isis interface ge-1/0/0 point-to-point
set protocols isis interface ge-3/0/0 point-to-point
set protocols isis address-family ipv6 unicast enable
set protocols isis address-family ipv6 unicast set interface ge-1/0/0
set protocols isis address-family ipv6 unicast set interface ge-3/0/0
set protocols isis level 2 address 49.0000.0000.0003.00

# 配置OSPFv3
set protocols ospf3 area 0.0.0.2 interface ge-2/0/0
set protocols ospf3 router-id 1.1.1.1

# 配置接口GigabitEthernet1/0/0
set interfaces ge-1/0/0 unit 0 family inet6 address 2001:db8:20::1/64
set interfaces ge-1/0/0 unit 0 family inet6 address 2001:db8:20::1/64 preferred

# 配置接口GigabitEthernet2/0/0
set interfaces ge-2/0/0 unit 0 family inet6 address 2001:db8:201::1/64
set interfaces ge-2/0/0 unit 0 family inet6 address 2001:db8:201::1/64 preferred

# 配置接口GigabitEthernet3/0/0
set interfaces ge-3/0/0 unit 0 family inet6 address 2001:db8:101::1/64
set interfaces ge-3/0/0 unit 0 family inet6 address 2001:db8:101::1/64 preferred
