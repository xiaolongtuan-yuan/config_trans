set system host-name LSRA

# MPLS 全局配置
set protocols mpls interface ge-1/0/0
set protocols mpls label-switched-path Tunnel10 to 2.2.2.2
set protocols mpls label-switched-path Tunnel10 bandwidth 1000
set protocols mpls traffic-engineering
set protocols mpls ldp interface lo0
set protocols mpls ldp router-id lo0.0

# L2VPN 配置
set protocols l2circuit neighbor 3.3.3.3 interface ge-1/0/0.1 virtual-circuit-id 30000 encapsulation-type vlan

# 接口配置
set interfaces ge-1/0/0 unit 0 family inet address 2.1.1.1/24
set interfaces ge-1/0/0 unit 1 vlan-id 1
set interfaces ge-1/0/0 unit 1 family inet
set interfaces ge-1/0/0 unit 1 family mpls

# Loopback 接口
set interfaces lo0 unit 0 family inet address 1.1.1.1/32

# Tunnel 接口
set interfaces gr-1/0/0 unit 0 tunnel source 1.1.1.1
set interfaces gr-1/0/0 unit 0 tunnel destination 2.2.2.2
set interfaces gr-1/0/0 unit 0 family inet
set interfaces gr-1/0/0 unit 0 family mpls

# 静态路由配置
set routing-options static route 2.2.2.2/32 next-hop gr-1/0/0.0
set routing-options static route 3.3.3.3/32 next-hop gr-1/0/0.0

# MPLS-TP 配置
set protocols oam mpls-tp connectivity-check interface gr-1/0/0 mep-id 1 remote-mep-id 2 l2circuit 30000 vlan

# 反向路径 (如果需要)
set protocols mpls label-switched-path Tunnel10-from-2_2_2_2 to 1.1.1.1
set protocols mpls label-switched-path Tunnel10-from-2_2_2_2 bandwidth 1000
