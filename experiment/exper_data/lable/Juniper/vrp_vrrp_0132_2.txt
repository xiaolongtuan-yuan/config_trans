# 设置设备主机名
set system host-name DeviceC

# 配置接口GigabitEthernet1/0/1
set interfaces ge-1/0/1 unit 0 family inet address 192.168.1.2/24
set interfaces ge-1/0/1 disable

# 配置接口GigabitEthernet1/0/2
set interfaces ge-1/0/2 unit 0 family inet address 192.168.2.2/24
set interfaces ge-1/0/2 disable

# 配置接口GigabitEthernet2/0/0
set interfaces ge-2/0/0 unit 0 family inet address 172.16.1.1/24
set interfaces ge-2/0/0 disable

# 启用OSPF协议
set protocols ospf area 0.0.0.0 interface ge-1/0/1.0
set protocols ospf area 0.0.0.0 interface ge-1/0/2.0
set protocols ospf area 0.0.0.0 interface ge-2/0/0.0
