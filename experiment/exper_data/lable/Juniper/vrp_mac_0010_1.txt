# 设置设备主机名
set system host-name CE

# 设置VLAN 2
set vlans vlan-2 vlan-id 2

# 创建MAC限制的安全策略（MAC access-list）
set firewall family ethernet-switching filter MAC-LIMIT term 1 then accept

# 配置接口GigabitEthernet1/0/1
set interfaces ge-1/0/1 unit 0 family ethernet-switching port-mode trunk
set interfaces ge-1/0/1 unit 0 family ethernet-switching vlan members 2
set interfaces ge-1/0/1 unit 0 ethernet-switching-options secure-access-port violation-action forward
set interfaces ge-1/0/1 ethernet-switching-options secure-access-port limit maximum 100

# 配置接口GigabitEthernet2/0/1
set interfaces ge-2/0/1 unit 0 family ethernet-switching port-mode trunk
set interfaces ge-2/0/1 unit 0 family ethernet-switching vlan members 2
set interfaces ge-2/0/1 ethernet-switching-options secure-access-port violation-action forward
set interfaces ge-2/0/1 ethernet-switching-options secure-access-port limit maximum 100
