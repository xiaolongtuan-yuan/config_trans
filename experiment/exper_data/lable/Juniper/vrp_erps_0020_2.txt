set system host-name DeviceC

set vlans vlan20 vlan-id 20
set vlans vlan300-400 vlan-id-list 300-400

set protocols rstp vlan 20 bridge-priority 0
set protocols rstp vlan 300-400 bridge-priority 0

set protocols erps instance 2 control-vlan 20
set protocols erps instance 2 protected-instance 2
set protocols erps instance 2 wtr-timer 6
set protocols erps instance 2 guard-timer 100
set protocols erps instance 2 version 2
set protocols erps instance 2 sub-ring

set interfaces ge-1/0/1 unit 0 family ethernet-switching vlan members [20 300-400]
set interfaces ge-1/0/1 unit 0 family ethernet-switching port-mode trunk
set interfaces ge-1/0/1 unit 0 family ethernet-switching storm-control default
set protocols erps instance 2 interface ge-1/0/1 owner

set interfaces ge-1/0/2 unit 0 family ethernet-switching vlan members [20 300-400]
set interfaces ge-1/0/2 unit 0 family ethernet-switching port-mode trunk
set interfaces ge-1/0/2 unit 0 family ethernet-switching storm-control default
set protocols erps instance 2 interface ge-1/0/2
