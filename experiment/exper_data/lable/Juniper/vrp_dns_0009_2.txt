set system host-name DeviceC

set interfaces ge-1/0/0 unit 0 family inet address 2.1.1.2/16
set interfaces ge-1/0/0 unit 0 description "GigabitEthernet1/0/0"

set interfaces ge-1/0/1 unit 0 family inet address 3.1.1.1/16
set interfaces ge-1/0/1 unit 0 description "GigabitEthernet1/0/1"

set interfaces lo0 unit 0 family inet address 4.1.1.2/32
set interfaces lo0 unit 0 description "Loopback0"

set protocols rip group RIP-GROUP export RIP-EXPORT
set protocols rip group RIP-GROUP neighbor ge-1/0/0.0
set protocols rip group RIP-GROUP neighbor ge-1/0/1.0
set protocols rip group RIP-GROUP neighbor lo0.0

set policy-options policy-statement RIP-EXPORT term 1 from protocol direct
set policy-options policy-statement RIP-EXPORT term 1 then accept
