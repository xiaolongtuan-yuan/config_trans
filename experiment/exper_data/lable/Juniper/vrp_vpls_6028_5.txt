# 设置设备名称
set system host-name CE2

# 配置物理接口
set interfaces ge-1/0/0 unit 0 family ethernet-switching port-mode trunk
set interfaces ge-1/0/0 unit 0 family ethernet-switching vlan members 10
set interfaces ge-1/0/0 unit 0 family ethernet-switching interface-modes access

# 配置子接口和IP地址
set interfaces ge-1/0/0 unit 10 vlan-id 10
set interfaces ge-1/0/0 unit 10 family inet address 10.1.1.2/24

# 启动接口
delete interfaces ge-1/0/0.10 disable
