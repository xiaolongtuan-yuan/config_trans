set system host-name Spoke1

set interfaces ge-1/0/0 unit 0 family inet address 10.1.2.10/24

set interfaces lo0 unit 0 family inet address 192.168.1.1/32

set interfaces gr-0/0/0 unit 0 tunnel source 10.1.2.10
set interfaces gr-0/0/0 unit 0 tunnel family inet address 172.16.1.2/24
set interfaces gr-0/0/0 unit 0 tunnel mode gre-multipoint

set security ipsec vpn PROFILE_NAME bind-interface gr-0/0/0.0
set security ipsec vpn PROFILE_NAME ike gateway GATEWAY_NAME
set security ipsec vpn PROFILE_NAME ike policy POLICY_NAME
set security ipsec vpn PROFILE_NAME establish-tunnels immediately

set protocols ospf area 0.0.0.0 interface gr-0/0/0.0
set protocols ospf area 0.0.0.0 interface lo0.0
set protocols ospf area 0.0.0.0 network 172.16.1.0/24
set protocols ospf area 0.0.0.0 network 192.168.1.0/24
set protocols ospf router-id 172.16.1.2

set protocols ospf2 area 1 interface ge-1/0/0.0
set protocols ospf2 area 1 network 10.1.1.0/24
set protocols ospf2 router-id 10.1.1.1
