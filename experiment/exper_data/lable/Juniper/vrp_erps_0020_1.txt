set system host-name DeviceB

set vlans vlan10 vlan-id 10
set vlans vlan100-200 vlan-id 100-200

set protocols rstp vlan 10 root-priority 0
set protocols rstp vlan 100-200 root-priority 0

set protocols erps instance 1 control-vlan 10
set protocols erps instance 1 protected-ring 1 instance 1
set protocols erps instance 1 wtr-timer 6000
set protocols erps instance 1 guard-timer 10000

set interfaces ge-1/0/1 unit 0 family ethernet-switching interface-mode trunk
set interfaces ge-1/0/1 unit 0 family ethernet-switching vlan members [10 100-200]
set interfaces ge-1/0/1 unit 0 family ethernet-switching erps ring 1 role owner

set interfaces ge-1/0/2 unit 0 family ethernet-switching interface-mode trunk
set interfaces ge-1/0/2 unit 0 family ethernet-switching vlan members [10 100-200]
set interfaces ge-1/0/2 unit 0 family ethernet-switching erps ring 1
