set system host-name Hub

set interfaces ge-1/0/0 unit 0 family inet address 10.1.1.10/24

set interfaces lo0 unit 0 family inet address 192.168.0.1/32

set interfaces gr-0/0/0 unit 0 family inet address 172.16.1.1/24
set interfaces gr-0/0/0 tunnel source 10.1.1.10
set interfaces gr-0/0/0 tunnel multipoint

set protocols bgp group BGP-PEERS type external
set protocols bgp group BGP-PEERS local-address 172.16.1.1
set protocols bgp group BGP-PEERS neighbor 172.16.1.2 peer-as 200
set protocols bgp group BGP-PEERS neighbor 172.16.1.3 peer-as 300
set protocols bgp group BGP-PEERS export EXPORT-OSPF
set protocols bgp network 192.168.0.0/16

set protocols ospf area 0.0.0.0 interface lo0.0
set protocols ospf area 0.0.0.0 interface ge-1/0/0.0
set protocols ospf area 0.0.0.0 interface gr-0/0/0.0

set protocols ospf3 area 0.0.0.1 interface ge-0/0/0.0
set protocols ospf3 router-id 10.1.1.10

set policy-options policy-statement EXPORT-OSPF term 1 from protocol ospf
set policy-options policy-statement EXPORT-OSPF term 1 then accept
