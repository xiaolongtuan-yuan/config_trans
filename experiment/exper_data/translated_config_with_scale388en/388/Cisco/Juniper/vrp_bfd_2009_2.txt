set protocols bfd all-interfaces enable
set vlans GigabitEthernet1/0/0 vlan-interfaces interface GigabitEthernet1/0/0
set vlans GigabitEthernet1/0/0 vlan-interfaces interface GigabitEthernet2/0/0

set interfaces [parameter1] ip address 192.168.2.2/255.255.255.0

commit and-quit
