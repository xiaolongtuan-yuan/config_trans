sysname Spoke2

nhrp enable
#
interface GigabitEthernet1/0/0
 ip address 10.1.3.10 255.255.255.0
 tunnel-protocol gre p2mp
 source GigabitEthernet2/0/0
 tunnel protection ipsec profile IPSEC_PROFILE_NAME
#
interface GigabitEthernet2/0/0
 ip address 192.168.2.1 255.255.255.0
# 
interface Tunnel0
 ip address 172.16.1.3 255.255.255.0
 tunnel-protocol gre
 source GigabitEthernet1/0/0
 destination 172.16.1.1
 tunnel protection ipsec profile IPSEC_PROFILE_NAME
# 
bgp 300
 router-id 172.16.1.3
 peer 172.16.1.1 as-number 100
 peer 172.16.1.2 as-number 200
 # 
 ipv4-family unicast
  undo synchronization
  import-route ospf 1
  peer 172.16.1.1 enable
  peer 172.16.1.2 enable
# 
ospf 1 router-id 172.16.1.3
 area 0.0.0.0
  network 192.168.2.0 0.0.0.255
# 
ospf 2 router-id 10.1.3.10
 area 0.0.0.1
  network 10.1.3.0 0.0.0.255
# 
return
