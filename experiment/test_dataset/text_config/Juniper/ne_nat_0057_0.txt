set system host-name NATA

set security zones security-zone trust interfaces ge-0/0/0 host-inbound-traffic system-services all
set security zones security-zone untrust interfaces ge-0/0/1 host-inbound-traffic system-services all

set security nat source pool address-group1 address 11.11.11.101 to 11.11.11.105

set security nat source rule-set OUTBOUND from zone trust
set security nat source rule-set OUTBOUND to zone untrust
set security nat source rule-set OUTBOUND rule NAT-INSIDE match source-address 192.168.10.0/24
set security nat source rule-set OUTBOUND rule NAT-INSIDE then source-nat pool address-group1
