set system host-name DeviceB

set interfaces ge-1/0/0 unit 0 family inet address 192.168.1.2/24
set interfaces ge-1/0/0 unit 0 description "GigabitEthernet1/0/0"

set interfaces ge-2/0/0 unit 0 family inet address 192.168.3.1/24
set interfaces ge-2/0/0 unit 0 description "GigabitEthernet2/0/0"

set interfaces ge-3/0/0 unit 0 family inet address 192.168.2.1/24
set interfaces ge-3/0/0 unit 0 description "GigabitEthernet3/0/0"

set routing-options router-id 2.2.2.2

set protocols bgp group internal neighbor 192.168.2.2
set protocols bgp group internal neighbor 192.168.3.2
