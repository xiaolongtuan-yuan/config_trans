# 设置设备的主机名
set system host-name CE4

# 配置GigabitEthernet接口
set interfaces ge-0/0/0 unit 0 family inet address 172.16.2.1/24

# 配置BGP
set protocols bgp group external-peers type external
set protocols bgp group external-peers peer-as 100
set protocols bgp group external-peers neighbor 172.16.2.2

# 启用BGP的IPv4单播地址族
set protocols bgp group external-peers family inet unicast

# 重新分发已连接的路由
set protocols bgp group external-peers export export-connected

# 配置不进行同步
set routing-options forwarding-table export export-connected

# 定义策略以重新分发已连接的路由到BGP
set policy-options policy-statement export-connected from protocol direct
set policy-options policy-statement export-connected then accept
