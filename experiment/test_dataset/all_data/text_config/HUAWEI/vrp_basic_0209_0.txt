
sysname SSH_Server
#
interface GigabitEthernet0/0/0
 undo shutdown
 ip address 10.1.1.2 255.255.0.0
#
rsa local-key-pair create 2048
#
pki realm TP
  certificate self-signed
  common-name SSH_Server
#
ssh server enable
ssh authentication-type default password
ssh user client001 authentication-type password
ssh user client001 service-type ssh
ssh user client001 level 3
#
ssh user client002 authentication-type publickey
ssh user client002 publickey sm2-key sm2key001
ssh user client002 service-type ssh
#
scp server enable
#
user-interface vty 0 4
 protocol inbound ssh
 authentication-mode aaa
#
return
