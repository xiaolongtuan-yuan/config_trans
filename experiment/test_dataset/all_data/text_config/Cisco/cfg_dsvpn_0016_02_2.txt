hostname Spoke2

! Interface Configuration
interface GigabitEthernet1/0/0
 ip address 10.1.3.10 255.255.255.0

interface Loopback0
 ip address 192.168.2.1 255.255.255.255

interface Tunnel0
 ip address 172.16.1.3 255.255.255.0
 tunnel source GigabitEthernet1/0/0
 tunnel key 1

! OSPF Configuration
router ospf 1
 router-id 172.16.1.3
 network 172.16.1.0 0.0.0.255 area 0
 network 192.168.2.0 0.0.0.255 area 0
 passive-interface GigabitEthernet1/0/0

! Crypto Configuration
crypto isakmp policy 1
 encryption aes
 authentication pre-share
 group 2
crypto isakmp key PRE_SHARED_KEY address 0.0.0.0 0.0.0.0

crypto ipsec transform-set IPSEC_TRANSFORM esp-aes esp-sha-hmac
 mode transport

crypto ipsec profile IPSEC_PROFILE
 set transform-set IPSEC_TRANSFORM

end
