hostname NATA
!
! NAT地址池配置
ip nat pool ADDRESS_GROUPA 11.1.1.1 11.1.1.10 netmask 255.255.255.0
ip nat pool ADDRESS_GROUPB 11.1.1.11 11.1.1.20 netmask 255.255.255.0
!

! 动态NAT配置（使用命名ACL）
ip access-list standard NAT_ACL_2464
ip access-list standard NAT_ACL_2465
!
ip nat inside source list NAT_ACL_2464 pool ADDRESS_GROUPA overload
ip nat inside source list NAT_ACL_2465 pool ADDRESS_GROUPB overload
!
! 内部接口配置
interface GigabitEthernet1/0/1
 no shutdown
 ip address 10.78.1.1 255.255.255.0
 ip nat inside
!
interface GigabitEthernet1/0/0
 no shutdown
 ip address 10.67.1.1 255.255.255.0
 ip nat inside
!
! 必须配置的外部接口（示例）
interface GigabitEthernet1/0/2
 no shutdown
 ip address 11.11.11.3 255.255.255.0
 ip nat outside
!
end
