hostname PE2
!
! MPLS基础配置
mpls ip
!mpls ldp
! router-id Loopback0
 !
 !address-family ipv4
 ! discovery targeted-hello accept
 !exit-address-family
!
! L2VPN伪线配置
pseudowire-class MPLS_PW
 encapsulation mpls
!
interface GigabitEthernet1/0/1.1
 encapsulation dot1Q 1
 xconnect 1.1.1.1 1001 pw-class MPLS_PW
!
interface GigabitEthernet1/0/2
 no shutdown
 ip address 192.168.1.1 255.255.255.0
 mpls ip
 mpls ldp ipv4
!
interface Loopback0
 ip address 2.2.2.2 255.255.255.255
!
! OSPF配置
router ospf 1
 network 2.2.2.2 0.0.0.0 area 0
 network 192.168.1.0 0.0.0.255 area 0
!
! Cisco标准CFM配置
ethernet cfm
 !domain MD1 level 3
  service MA1
  ! continuity-check
   !mep 2 interface GigabitEthernet1/0/1.1 inward
   !
   !remote-mep 1
   ! continuity-check
!
! 延迟测量配置
ethernet cfm
 !domain MD1 level 3
  service MA1
  ! delay-measurement
   ! one-way receive test-id 1
    !one-way continual send test-id 3 interval 1000
!
end
