hostname DeviceA

! 全局L2TP启用
vpdn enable
!

! AAA配置
aaa new-model
aaa authentication ppp default group radius
aaa authorization network default group radius
aaa accounting network default start-stop group radius
!

! Radius服务器配置
radius-server host 10.0.0.249 auth-port 1812 acct-port 1813
radius-server key 7 050803150C1E
!

! VRF配置
vrf definition vrf1
 rd 100:1
 route-target export 100:1
 route-target import 100:1
!

! 虚拟接口模板
interface Virtual-Template1
 ppp authentication chap
!

!

interface GigabitEthernet2/0/0.100
 encapsulation dot1Q 100
!

! L2TP隧道配置
vpdn-group LAC1
 accept-dialin
  protocol l2tp
  virtual-template 1
 l2tp tunnel password 7 08090A1B2C3D4E
!

! 接口配置
interface Loopback0
 vrf forwarding vrf1
 ip address 1.1.1.1 255.255.255.255
!

interface GigabitEthernet1/0/1
 no shutdown
 vrf forwarding vrf1
 ip address 10.0.0.1 255.255.255.0
!

! 路由配置
ip route vrf vrf1 3.3.3.3 255.255.255.255 10.0.0.2
!
