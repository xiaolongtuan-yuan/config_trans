hostname NAT64_CGN

! License activation (示例语法)
license boot module c1900 technology-package securityk9

! NAT全局参数
ip nat translation timeout 86400
ip nat translation tcp-timeout 86400
ip nat translation udp-timeout 300
ip nat translation icmp-timeout 10

! IPv6 ACL定义
ipv6 access-list NAT64_V6_ACL
 permit ipv6 2001:DB8::1:1112/126 64:FF9B::/96

! IPv4 ACL定义
ip access-list standard NAT64_V4_ACL
 permit 11.1.1.0 0.0.0.255

! NAT64配置
!nat64 enable
!nat64 prefix stateful 64:FF9B::/96
!nat64 v4 pool NAT64_POOL 11.1.1.1 11.1.1.255 prefix-length 24
!nat64 route 64:FF9B::/96 GigabitEthernet1/0/1

! 接口配置
interface GigabitEthernet1/0/1
 ipv6 address 2001:DB8::1:110E/126

! NAT重载配置
ip nat inside source list NAT64_V4_ACL interface GigabitEthernet1/0/1 overload

! 日志配置
logging host 1.1.1.1
logging source-interface GigabitEthernet1/0/1
logging trap debugging

end
