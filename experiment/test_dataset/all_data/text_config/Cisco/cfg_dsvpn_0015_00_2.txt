hostname Spoke2
!
! 修正NHRP配置（无需全局启用，接口级配置即可）
!
interface GigabitEthernet1/0/0
 ip address 10.1.3.10 255.255.255.0
!
interface LoopBack0
 ip address 192.168.2.1 255.255.255.255
!
! 修正Tunnel接口配置（使用标准DMVPN配置）
interface Tunnel0
 ip address 172.16.1.3 255.255.255.0
 tunnel source GigabitEthernet1/0/0
 tunnel mode gre multipoint  ! 部分平台可能支持，若不支持需改为ipsec配置
 tunnel key 1
 ip nhrp network-id 1
 ip nhrp nhs 172.16.1.1 nbma 10.1.1.10 multicast
 ip nhrp map multicast dynamic
 ip nhrp shortcut
!
router ospf 2
 router-id 10.1.3.10
 network 10.1.3.0 0.0.0.255 area 0.0.0.1
 network 172.16.1.0 0.0.0.255 area 0.0.0.1
!
ip route 192.168.0.0 255.255.255.0 Tunnel0
ip route 192.168.1.0 255.255.255.0 Tunnel0
!
end
