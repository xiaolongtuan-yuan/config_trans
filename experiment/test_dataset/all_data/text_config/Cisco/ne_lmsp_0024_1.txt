hostname IS-IS-Router
!
router isis 1
 is-type level-1
 net 49.0022.0022.0022.0022.00
 !
 address-family ipv4
  metric-style wide  ! Cisco标准metric样式
  mpls traffic-eng level-1-2  ! TE在地址族下配置
  spf-interval 1 50  ! 修正SPF间隔语法
 exit-address-family
!
interface GigabitEthernet2/0/1
 ip router isis 1
 isis circuit-type level-1  ! 明确指定电路类型
!
interface GigabitEthernet2/0/5
 ip router isis 1
 isis circuit-type level-1
!
interface Loopback0
 ip address 2.2.2.2 255.255.255.255
 ip router isis 1
!
end
