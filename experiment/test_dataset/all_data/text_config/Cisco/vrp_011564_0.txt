  service MA1 vlan 2
 !  continuity-check
  ! mep 2 interface GigabitEthernet1/0/1 outward

  service evc MA1

 no shutdown

!

  service MA1
  ! remote-mep 1
   ! continuity-check
    !loss-threshold 3
