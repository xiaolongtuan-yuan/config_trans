hostname PE1
!
! 修正CFM配置（Cisco标准语法）
ethernet cfm global
ethernet cfm domain md1 level 3
!
! MPLS基础配置
!mpls label protocol ldp explicit-null
mpls ldp router-id Loopback0 force
mpls ip
!
! 修正L2VPN配置
l2vpn bridge group LDP1
 bridge-domain LDP1
!
! 子接口配置修正
interface GigabitEthernet1/0/1.1
 no ip address
 encapsulation dot1Q 2
 service instance 2 ethernet
  rewrite ingress tag pop 1 symmetric
  bridge-domain LDP1
!
! 物理接口配置
interface GigabitEthernet1/0/2
 no shutdown
 ip address 10.1.1.1 255.255.255.0
 mpls ip
!
! Loopback接口修正（改为/32掩码）
interface Loopback0
 ip address 2.2.2.2 255.255.255.255
!
! OSPF配置
router ospf 1
 network 2.2.2.2 0.0.0.0 area 0
 network 10.1.1.0 0.0.0.255 area 0
!
! 修正CFM MEP配置（Cisco标准语法）
ethernet cfm service ma1 vlan 2
 !continuity-check
!
ethernet cfm mep 1 domain md1 service ma1
 ethernet cfm mep 1 domain md1 service ma1 inward
 ethernet cfm mep 1 domain md1 service ma1 continuity-check interval 3s
!
ethernet cfm mep crosscheck
 ethernet cfm mep crosscheck start-delay 30
 ethernet cfm mep remote 2 domain md1 service ma1
!
ethernet cfm loss-measurement
 !mep 1 domain md1 service ma1
 !measurement-interval 100ms
 !dual-ended
!
