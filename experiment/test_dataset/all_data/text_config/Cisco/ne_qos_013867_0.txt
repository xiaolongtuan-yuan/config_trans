hostname Router
!
ip multicast-routing
!
ip access-list standard huawei-qos-acl
 permit 239.0.0.0 0.0.0.255
!
class-map match-any MULTICAST-QOS
 match access-group name huawei-qos-acl
!
policy-map MULTICAST-QOS-POLICY
 class MULTICAST-QOS
  bandwidth remaining percent 100
!
interface GigabitEthernet1/0/0
 no shutdown
 ip pim sparse-mode
 ip igmp version 3
!
interface GigabitEthernet1/0/0.100
 encapsulation dot1Q 100
 ip pim sparse-mode
 ip igmp version 3
 service-policy input MULTICAST-QOS-POLICY
!
interface GigabitEthernet2/0/0
 no shutdown
 ip address 10.0.0.3 255.255.255.0
 ip pim sparse-mode
!
