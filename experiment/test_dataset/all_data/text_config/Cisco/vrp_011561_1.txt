hostname PE2
!
! CFM配置（标准Cisco语法）
ethernet cfm ieee
ethernet cfm domain md1 level 3
 service ma1 evc EVPN-A vlan 1
  !continuity-check interval 10
  !continuity-check
!
ethernet cfm mep 2 domain md1 service ma1
 !interface GigabitEthernet1/0/1 in
 !continuity-check
!
ethernet cfm remote-mep 1 domain md1 service ma1
 !continuity-check
!
ethernet cfm sla 1 mep 2 domain md1 service ma1
 !delay-measurement one-way receive
!
ethernet cfm sla 3 mep 2 domain md1 service ma1
 !delay-measurement one-way continuous interval 1000
!
! EVPN配置
l2vpn evpn instance 1 vlan-based
 !rd 200:1
 !route-target both 1:1
!
mpls ldp router-id 2.2.2.2 force
!
interface GigabitEthernet1/0/1
 no shutdown
 service instance 1 ethernet
  encapsulation dot1q 1
  rewrite ingress tag pop 1 symmetric
  bridge-domain 1
!
interface GigabitEthernet1/0/2
 no shutdown
 ip address 10.2.1.1 255.255.255.0
 mpls ip
 mpls ldp
!
interface Loopback0
 ip address 2.2.2.2 255.255.255.255
!
router bgp 100
 bgp log-neighbor-changes
 neighbor 1.1.1.1 remote-as 100
 neighbor 1.1.1.1 update-source Loopback0
 !
 !address-family l2vpn evpn
  neighbor 1.1.1.1 activate
  neighbor 1.1.1.1 send-community extended
 !exit-address-family
!
router ospf 1
 router-id 2.2.2.2
 network 2.2.2.2 0.0.0.0 area 0
 network 10.2.1.0 0.0.0.255 area 0
!
