hostname PE1
!
! 移除非标准CFM配置，改用Cisco标准Ethernet CFM
ethernet cfm global
!
! MPLS基础配置
mpls ip
mpls ldp router-id Loopback1 force
!
! 目标LDP邻居配置
mpls ldp neighbor 3.3.3.3 targeted
!
! 接口配置
interface GigabitEthernet1/0/1
 no shutdown
!
interface GigabitEthernet1/0/1.1
 encapsulation dot1Q 10
 service instance 10 ethernet
  rewrite ingress tag pop 1 symmetric
  xconnect 3.3.3.3 100 encapsulation mpls
!
interface GigabitEthernet1/0/2
 no shutdown
 ip address 10.1.1.1 255.255.255.252
 mpls ip
!
interface Loopback1
 ip address 1.1.1.1 255.255.255.255
!
! OSPF配置
router ospf 1
 network 1.1.1.1 0.0.0.0 area 0
 network 10.1.1.0 0.0.0.3 area 0
!
! 修正CFM配置（Cisco标准语法）
ethernet cfm domain md1 level 3
ethernet cfm service ma1 vlan 10
!
! MEP配置
ethernet cfm mep 31 domain md1 service ma1
 ethernet cfm mep 31 domain md1 service ma1 inward
 ethernet cfm mep 31 domain md1 service ma1 continuity-check
!
! 远程MEP配置
ethernet cfm mep crosscheck
 ethernet cfm mep crosscheck start-delay 30
 ethernet cfm mep remote 32 domain md1 service ma1
!
end
