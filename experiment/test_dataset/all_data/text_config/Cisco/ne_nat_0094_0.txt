hostname BRAS

interface GigabitEthernet2/0/0.1
 encapsulation dot1q 1
 ip address 10.10.10.1 255.255.255.0
 ip nat inside
 ip virtual-reassembly
!

ip dhcp pool pool1
 network 10.10.10.0 255.255.255.0
 default-router 10.10.10.1
 dns-server 192.168.7.252
!

ip nat pool pool1 11.11.11.101 11.11.11.105 netmask 255.255.255.0
ip nat inside source list ACL-3001 pool pool1 overload
!

ip access-list extended ACL-3001
 permit ip 10.10.10.0 0.0.0.255 any
!

ip access-list extended ACL-6001
 permit ip any any  ! Modified since object-group is not standard
!

class-map match-all CM-c1
 match access-group name ACL-6001
!

policy-map PM-p1
 class CM-c1
  police cir 1000000
   conform-action transmit
   exceed-action drop
!

interface GigabitEthernet2/0/0.1
 service-policy input PM-p1
!

aaa new-model
aaa authentication dot1x default group radius
aaa authorization network default group radius
aaa accounting network default start-stop group radius
!

radius-server host 192.168.7.252 auth-port 1812 acct-port 1813 key 0 <shared_secret>
!

interface GigabitEthernet2/0/0.1
 dot1x pae authenticator
 authentication port-control auto
!

radius-server deadtime 5
radius-server retransmit 3
!
end
