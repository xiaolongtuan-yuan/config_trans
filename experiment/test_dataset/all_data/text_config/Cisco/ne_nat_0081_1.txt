hostname CGNA
!
! 使用标准扩展ACL替代编号ACL
ip access-list extended NAT-TRAFFIC
 permit ip 192.168.10.0 0.0.0.255 any
!
! NAT地址池配置
ip nat pool POOL1 11.11.11.1 11.11.11.254 netmask 255.255.255.0
!
! NAT规则配置
ip nat inside source list NAT-TRAFFIC pool POOL1 overload
!
! 内部接口配置
interface GigabitEthernet2/0/0
 ip address 192.168.10.1 255.255.255.0
 ip nat inside
 no shutdown
!
! 外部接口配置
interface GigabitEthernet2/0/1
 ip address 11.11.11.2 255.255.255.0
 ip nat outside
 no shutdown
!
end
