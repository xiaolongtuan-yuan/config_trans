hostname NATA

! 移除无效的全局NAT启用命令
! ip nat enable (已移除)

! 配置NAT地址池和ACL
ip nat pool address-group1 11.11.11.101 11.11.11.105 netmask 255.255.255.0
ip nat inside source list 3001 pool address-group1 overload

ip access-list extended 3001
 permit ip 192.168.10.0 0.0.0.255 any

! 配置接口NAT方向
interface GigabitEthernet1/0/0
 no shutdown
 ip address 192.168.10.1 255.255.255.0
 ip nat inside

interface GigabitEthernet1/0/1
 no shutdown
 ip address 1.1.1.1 255.255.255.0
 ip nat outside

! 修正静态NAT映射（原配置有误）
ip nat inside source static 192.168.10.100 11.11.11.101

end
