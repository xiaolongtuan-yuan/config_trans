hostname DeviceA
!
ip access-list extended ACL_3001
 permit ip host 10.100.0.11 any
!
ip access-list extended ACL_3002
 permit ip host 10.110.0.11 any
!
interface Tunnel30
!
interface Tunnel40
!
route-map PBR_AAA permit 5
 match ip address ACL_3001
 set ip next-hop 10.100.0.1  ! 必须指定具体下一跳
!
route-map PBR_AAA permit 10
 match ip address ACL_3002
 set ip next-hop 10.110.0.1  ! 必须指定具体下一跳
!
interface GigabitEthernet1/0/0
 no shutdown
 ip address 10.100.0.10 255.255.255.0
 ip policy route-map PBR_AAA  ! 修正策略路由语法
!
interface GigabitEthernet2/0/0
 no shutdown
 ip address 10.110.0.10 255.255.255.0
 ip policy route-map PBR_AAA  ! 修正策略路由语法
!
end
