set system host-name Device

# RADIUS 服务器配置（保持不变）
set access radius-server 10.20.20.1 secret "$9$vS%796FO7%C~pB%CR=q;j}gSCqR-X6+P!.DYI@)"  # 加密格式
set access radius-server 10.20.20.1 port 1812
set access radius-server 10.20.20.1 accounting-port 1813

# 访问配置文件
set access profile doma1 authentication-order radius
set access profile doma1 accounting order radius
set access profile doma1 address-assignment pool pool1
set access profile doma1 accounting login-on-success

# QoS 配置修正
set class-of-service schedulers QoS-Pro1 transmit-rate percent 10  # 百分比或明确单位
set class-of-service schedulers QoS-Pro1 buffer-size percent 30
set class-of-service forwarding-classes class default scheduler QoS-Pro1

# 地址池配置（保持不变）
set access address-assignment pool pool1 family inet network 10.10.10.0/24
set access address-assignment pool pool1 family inet range pool-range low 10.10.10.2
set access address-assignment pool pool1 family inet range pool-range high 10.10.10.100
set routing-options static route 10.10.10.0/24 next-hop 10.10.10.1

# 接口配置修正
set interfaces ge-1/0/0 flexible-vlan-tagging
set interfaces ge-1/0/0 unit 0 family ethernet-switching interface-mode access
set interfaces ge-1/0/0 unit 0 family ethernet-switching vlan members 270


# Loopback 配置
set interfaces lo0 unit 0 family inet address 192.168.0.1/32

# L2TP 配置修正
set services l2tp access-profile doma1
set services l2tp tunnel LNS local-address 192.168.0.1
set services l2tp tunnel LNS authentication password "$9$EncryptedPassword"  # 加密格式
