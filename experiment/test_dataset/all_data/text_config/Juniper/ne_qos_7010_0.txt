set system host-name DeviceA

# Firewall filter for PBR
set firewall family inet filter PBR_aaa term match_3001 from source-address 10.100.0.11/32
set firewall family inet filter PBR_aaa term match_3001 then routing-instance INSTANCE_TUNNEL30
set firewall family inet filter PBR_aaa term match_3001 then accept

set firewall family inet filter PBR_aaa term match_3002 from source-address 10.110.0.11/32
set firewall family inet filter PBR_aaa term match_3002 then routing-instance INSTANCE_TUNNEL40
set firewall family inet filter PBR_aaa term match_3002 then accept

set firewall family inet filter PBR_aaa term default then accept

# GRE Tunnel 30 Configuration
set interfaces gr-0/0/30 unit 0 tunnel source <local-gre-source-IP>
set interfaces gr-0/0/30 unit 0 tunnel destination <remote-tunnel30-IP>
set interfaces gr-0/0/30 unit 0 family inet mtu 1500

# GRE Tunnel 40 Configuration
set interfaces gr-0/0/40 unit 0 tunnel source <local-gre-source-IP>
set interfaces gr-0/0/40 unit 0 tunnel destination <remote-tunnel40-IP>
set interfaces gr-0/0/40 unit 0 family inet mtu 1500

# Interface configurations with PBR
set interfaces ge-0/0/0 unit 0 family inet address 10.100.0.10/24
set interfaces ge-0/0/0 unit 0 family inet filter input PBR_aaa

set interfaces ge-0/0/1 unit 0 family inet address 10.110.0.10/24
set interfaces ge-0/0/1 unit 0 family inet filter input PBR_aaa

# Routing instances for tunnel interfaces
set routing-instances INSTANCE_TUNNEL30 instance-type forwarding
set routing-instances INSTANCE_TUNNEL30 interface gr-0/0/30.0

set routing-instances INSTANCE_TUNNEL40 instance-type forwarding
set routing-instances INSTANCE_TUNNEL40 interface gr-0/0/40.0
