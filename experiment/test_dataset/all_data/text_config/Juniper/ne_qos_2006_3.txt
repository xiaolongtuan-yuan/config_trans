set system host-name CE1

# 接口配置修正
set interfaces ge-0/0/0 unit 0 family inet address 10.1.1.1/24

# BGP 配置
set routing-options autonomous-system 65410

set protocols bgp group EBGP neighbor 10.1.1.2 peer-as 100
set protocols bgp group EBGP type external
set protocols bgp group EBGP family inet unicast
set protocols bgp group EBGP export EXPORT-ROUTES
set protocols bgp group EBGP import ACCEPT-ROUTES

# 路由策略
set policy-options policy-statement EXPORT-ROUTES term 1 from route-filter 10.1.1.0/24 exact
set policy-options policy-statement EXPORT-ROUTES term 1 then accept

set policy-options policy-statement ACCEPT-ROUTES then accept
