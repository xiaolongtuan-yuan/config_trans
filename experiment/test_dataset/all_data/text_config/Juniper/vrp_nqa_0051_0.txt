set system host-name DeviceA

set vlans vlan-10 vlan-id 10

set services rpm probe sla-test test icmp-test source-interface ge-1/0/1
set services rpm probe sla-test test icmp-test target address 10.1.1.1
set services rpm probe sla-test test icmp-test probe-count 2
set services rpm probe sla-test test icmp-test probe-interval 5
set services rpm probe sla-test test icmp-test test-interval 3

set interfaces ge-1/0/1 unit 0 family ethernet-switching port-mode trunk
set interfaces ge-1/0/1 unit 0 family ethernet-switching vlan members vlan-10
set interfaces ge-1/0/1 enable

set interfaces ge-1/0/2 unit 0 family ethernet-switching port-mode trunk
set interfaces ge-1/0/2 unit 0 family ethernet-switching vlan members vlan-10
set interfaces ge-1/0/2 enable
