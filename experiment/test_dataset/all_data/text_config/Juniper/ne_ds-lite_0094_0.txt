set system host-name DS-Lite

# IPv6防火墙过滤器（保持不变）
set firewall family inet6 filter DS-Lite-ACL term 1 from source-address 2001:DB8:2::/64
set firewall family inet6 filter DS-Lite-ACL term 1 then accept

# 接口配置修正
set interfaces ge-0/0/1 unit 0 family inet6 filter input DS-Lite-ACL
set interfaces ge-0/0/1 unit 0 family inet6 mtu 1500

# 路由实例配置修正
set routing-instances DS-Lite1 instance-type virtual-router
set routing-instances DS-Lite1 route-distinguisher 100:1
set routing-instances DS-Lite1 vrf-target target:100:1

# 隧道接口配置修正
set interfaces gr-0/0/0 unit 0 tunnel source ge-0/0/1.0
set interfaces gr-0/0/0 unit 0 tunnel destination 2001:DB8:2::1
set interfaces gr-0/0/0 unit 0 family inet6
set interfaces gr-0/0/0 unit 0 tunnel encapsulation ipip6
set interfaces gr-0/0/0 unit 0 tunnel ttl 255

# IS-IS配置修正
set protocols isis interface ge-0/0/1.0
set protocols isis level 1 disable
set protocols isis level 2 disable
