set system host-name JUNIPER-QOS

# 多播基础配置
set routing-options multicast asm-override-ssm
set protocols pim rp local address 10.0.0.1
set protocols pim interface ge-2/0/0.0 mode sparse

# IGMP配置
set protocols igmp interface ge-1/0/0.0 version 3
set protocols igmp interface ge-1/0/0.100 version 3
set protocols igmp-snooping vlan all

# 接口配置
set interfaces ge-1/0/0 unit 0 family ethernet-switching
set interfaces ge-1/0/0 unit 100 vlan-id 100
set interfaces ge-1/0/0 unit 100 family inet address 192.168.100.1/24
set interfaces ge-2/0/0 unit 0 family inet address 10.0.0.3/24

set class-of-service forwarding-classes queue 3 best-effort
set class-of-service forwarding-classes queue 2 expedited-forwarding
set class-of-service forwarding-classes queue 1 assured-forwarding
set class-of-service forwarding-classes queue 0 network-control

set firewall family inet filter MULTICAST-QOS term CLASSIFY-MCAST from destination-address 239.0.0.0/24
set firewall family inet filter MULTICAST-QOS term CLASSIFY-MCAST then forwarding-class expedited-forwarding
set firewall family inet filter MULTICAST-QOS term CLASSIFY-MCAST then accept
set firewall family inet filter MULTICAST-QOS term DEFAULT then accept

set class-of-service schedulers EF-SCHEDULER transmit-rate percent 100
set class-of-service schedulers EF-SCHEDULER buffer-size percent 100
set class-of-service scheduler-maps QOS-MAP forwarding-class expedited-forwarding scheduler EF-SCHEDULER

set class-of-service interfaces ge-1/0/0 unit 100 scheduler-map QOS-MAP
set interfaces ge-1/0/0 unit 100 family inet filter input MULTICAST-QOSz
