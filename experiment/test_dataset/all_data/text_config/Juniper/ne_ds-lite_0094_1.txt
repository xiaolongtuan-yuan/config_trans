set system host-name DS-Lite

# IPv6 Firewall Filters
set firewall family inet6 filter 3500 term 1 from source-address 2001:DB8:2::/64
set firewall family inet6 filter 3500 term 1 then accept

# DS-Lite Routing Instance
set routing-instances DS-Lite instance-type virtual-router
set routing-instances DS-Lite routing-options router-id 1.1.1.1
set routing-instances DS-Lite interface ge-0/0/2.0

# Global Routing Options
set routing-options rib inet6.0

# Interface Configuration
set interfaces ge-0/0/2 unit 0 family inet6 filter input p1


# Firewall Filter with Routing Action (Corrected)
set firewall family inet6 filter p1 term 1 from source-address 2001:DB8:2::/64
set firewall family inet6 filter p1 term 1 then routing-instance DS-Lite

# ISIS Protocol Configuration (Corrected)
set protocols isis interface ge-0/0/2.0 level 1 metric 10

# Export Policies
set policy-options policy-statement connected-to-isis term 1 from protocol direct
set policy-options policy-statement connected-to-isis term 1 then accept
set policy-options policy-statement static-to-isis term 1 from protocol static
set policy-options policy-statement static-to-isis term 1 then accept
