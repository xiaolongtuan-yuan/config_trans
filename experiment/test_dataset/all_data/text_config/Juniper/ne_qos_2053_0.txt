set system host-name DeviceA

# Firewall Policer Configuration
set firewall policer p1 if-exceeding bandwidth-limit 10m
set firewall policer p1 if-exceeding burst-size-limit 150k
set firewall policer p1 then discard

# Firewall Filter Configuration
set firewall filter f1 term t1 then policer p1
set firewall filter f1 term t1 then accept

# Interface Configuration - ge-0/1/0
set interfaces ge-0/1/0 unit 0 family inet address 10.10.1.1/24
set interfaces ge-0/1/0 disable

# Interface Configuration - ge-0/2/0
set interfaces ge-0/2/0 flexible-vlan-tagging
set interfaces ge-0/2/0 unit 0 vlan-id 1
set interfaces ge-0/2/0 unit 0 family inet address 10.1.1.1/24
set interfaces ge-0/2/0 unit 0 family inet filter input f1
set interfaces ge-0/2/0 unit 100 vlan-id 100

# OSPF Configuration
set protocols ospf area 0 interface ge-0/1/0.0
set protocols ospf area 0 interface ge-0/2/0.0
