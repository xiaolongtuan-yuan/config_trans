set system host-name NAT64_CGN
set system syslog host 1.1.1.1 port 514
set system syslog host 1.1.1.1 any info


set security policies from-zone trust to-zone untrust policy IPV6_TO_NAT64 match source-address IPV6_HOST_1
set security policies from-zone trust to-zone untrust policy IPV6_TO_NAT64 match destination-address NAT64_PREFIX
set security policies from-zone trust to-zone untrust policy IPV6_TO_NAT64 then permit

set security nat source pool NAT64_POOL address 11.1.1.1/24

set security nat source rule-set NAT64_SOURCE rule RULE1 then source-nat pool NAT64_POOL

set security nat destination pool NAT64_DEST_POOL address 11.1.1.1/24

set security nat destination rule-set NAT64_DEST rule RULE1 then destination-nat pool NAT64_DEST_POOL

set applications application TCP_APP protocol tcp
set applications application TCP_APP inactivity-timeout 86400
set applications application UDP_APP protocol udp
set applications application UDP_APP inactivity-timeout 1200
set applications application ICMP_APP protocol icmp
set applications application ICMP_APP inactivity-timeout 10

set security flow tcp-session timeout 86400
set security flow udp-session timeout 1200
set security flow icmp-session timeout 10
