set system host-name DeviceA

# IPv6 配置部分
set interfaces ge-0/0/1 unit 0 family inet address 10.10.10.1/24

# 接口 ge-0/0/2 配置
set interfaces ge-0/0/2 flexible-vlan-tagging
set interfaces ge-0/0/2 encapsulation flexible-ethernet-services
set interfaces ge-0/0/2 unit 0 family inet6
set interfaces ge-0/0/2 unit 1 family inet address 10.0.0.1/24
set interfaces ge-0/0/2 unit 1 encapsulation vlan-bridge
set interfaces ge-0/0/2 unit 1 vlan-id 1

# 接口 ge-0/0/3 配置
set interfaces ge-0/0/3 unit 0 family inet address 192.168.1.1/24

# 路由配置
set routing-options static route 10.1.1.1/32 next-hop 10.0.0.2
set routing-options rib inet6.0 static route 2001:db8::/128 next-hop 2001:db8:2::2

# DHCPv4 中继配置
set forwarding-options dhcp-relay server-group v4-servers 192.168.1.2
set forwarding-options dhcp-relay active-server-group v4-servers
