set system host-name CE2

set vlans vlan2 vlan-id 2

# CFM Configuration
set protocols oam ethernet connectivity-fault-management
set protocols oam ethernet connectivity-fault-management maintenance-domain md3 level 0
set protocols oam ethernet connectivity-fault-management maintenance-domain md3 maintenance-association ma3 vlan vlan2

# MEP Configuration
set protocols oam ethernet connectivity-fault-management maintenance-domain md3 maintenance-association ma3 mep 4 interface ge-1/0/1.0 direction down
set protocols oam ethernet connectivity-fault-management maintenance-domain md3 maintenance-association ma3 mep 4 continuity-check transmit interval 1s
set protocols oam ethernet connectivity-fault-management maintenance-domain md3 maintenance-association ma3 remote-mep 3

# Loss Measurement Configuration
set protocols oam ethernet connectivity-fault-management maintenance-domain md3 maintenance-association ma3 mep 4 loss-measurement single-ended receive

# Interface Configuration
set interfaces ge-1/0/1 flexible-vlan-tagging
set interfaces ge-1/0/1 disable
set interfaces ge-1/0/1 unit 0 family ethernet-switching interface-mode trunk
set interfaces ge-1/0/1 unit 0 family ethernet-switching vlan members vlan2
