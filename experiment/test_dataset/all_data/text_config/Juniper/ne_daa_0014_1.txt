set system host-name JUNIPER

# Firewall Filters
set firewall family inet filter 3001 term 1 from source-address 10.11.11.1/32
set firewall family inet filter 3001 term 1 then accept
set firewall family inet filter 3001 term 2 from source-address 10.11.11.2/32
set firewall family inet filter 3001 term 2 then accept

set firewall family inet filter 3002 term 1 from source-address 10.11.11.3/32
set firewall family inet filter 3002 term 1 then accept

# Policers Configuration (Corrected syntax)
set firewall policer policer_3001 if-exceeding bandwidth-limit 1m burst-size-limit 1m
set firewall policer policer_3001 then discard

set firewall policer policer_3002 if-exceeding bandwidth-limit 1m burst-size-limit 1m
set firewall policer policer_3002 then discard

# Interface Configuration
set interfaces ge-0/0/1 unit 0 family inet address 192.168.1.1/24
set interfaces ge-0/0/1 unit 0 family inet filter input 3001
set interfaces ge-0/0/1 unit 0 family inet filter output 3002

# Apply Policers to Interface (Corrected syntax)
set interfaces ge-0/0/1 unit 0 family inet policer input policer_3001
set interfaces ge-0/0/1 unit 0 family inet policer output policer_3002
