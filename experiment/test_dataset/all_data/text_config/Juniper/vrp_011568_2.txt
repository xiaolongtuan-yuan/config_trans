set system host-name CE1

# VLAN配置
set vlans vlan10 vlan-id 10

# 接口配置
set interfaces ge-0/0/1 unit 0 family ethernet-switching interface-mode trunk
set interfaces ge-0/0/1 unit 0 family ethernet-switching vlan members vlan10

# CFM配置（修正语法）
set protocols oam ethernet connectivity-fault-management
set protocols oam ethernet connectivity-fault-management maintenance-domain md1 level 6
set protocols oam ethernet connectivity-fault-management maintenance-domain md1 maintenance-association ma1 vlan vlan10
set protocols oam ethernet connectivity-fault-management maintenance-domain md1 maintenance-association ma1 continuity-check interval 1s

# MEP配置（维护端点）
set protocols oam ethernet connectivity-fault-management maintenance-domain md1 maintenance-association ma1 mep 1 interface ge-0/0/1.0
set protocols oam ethernet connectivity-fault-management maintenance-domain md1 maintenance-association ma1 mep 1 direction outward
set protocols oam ethernet connectivity-fault-management maintenance-domain md1 maintenance-association ma1 mep 1 continuity-check

# 远程MEP配置
set protocols oam ethernet connectivity-fault-management maintenance-domain md1 maintenance-association ma1 remote-mep 2
