set system host-name SRX-NAT-Gateway

# 接口和安全区域配置
set interfaces ge-0/0/0 unit 0 family inet address 10.1.1.1/24
set security zones security-zone trust interfaces ge-0/0/0.0

set interfaces ge-0/0/1 unit 0 family inet address 11.11.11.101/24
set security zones security-zone untrust interfaces ge-0/0/1.0

set security nat static rule-set STATIC-NETWORK from zone trust
set security nat static rule-set STATIC-NETWORK rule SUBNET-TRANSLATE match source-address 10.0.0.0/24
set security nat static rule-set STATIC-NETWORK rule SUBNET-TRANSLATE then static-nat prefix 11.11.11.0/24

set security nat static rule-set PORT-MAPPING-256 from zone trust
set security nat static rule-set PORT-MAPPING-256 rule TCP-256 match source-address 10.0.0.0/24
set security nat static rule-set PORT-MAPPING-256 rule TCP-256 match destination-port 256
set security nat static rule-set PORT-MAPPING-256 rule TCP-256 then static-nat prefix 11.11.11.101/32
!set security nat static rule-set PORT-MAPPING-256 rule TCP-256 then static-nat mapped-port 256

set security nat static rule-set PORT-MAPPING-1023 from zone trust
set security nat static rule-set PORT-MAPPING-1023 rule TCP-1023 match source-address 10.0.0.0/24
set security nat static rule-set PORT-MAPPING-1023 rule TCP-1023 match destination-port 1023
set security nat static rule-set PORT-MAPPING-1023 rule TCP-1023 then static-nat prefix 11.11.11.101/32
!set security nat static rule-set PORT-MAPPING-1023 rule TCP-1023 then static-nat mapped-port 1023
