set system host-name SPE

# VRF Configuration
set routing-instances vpn1 instance-type vrf
set routing-instances vpn1 route-distinguisher 100:1
set routing-instances vpn1 vrf-target target:100:1

# LDP Configuration
set protocols ldp interface lo0.0
set protocols ldp interface ge-1/0/0.0
set protocols ldp interface ge-2/0/0.0

# MPLS Configuration
set protocols mpls interface ge-1/0/0.0
set protocols mpls interface ge-2/0/0.0

# Interface Configuration
set interfaces ge-1/0/0 unit 0 family inet address 10.1.1.2/24
set interfaces ge-1/0/0 unit 0 family mpls

set interfaces ge-2/0/0 unit 0 family inet address 10.2.1.1/24
set interfaces ge-2/0/0 unit 0 family mpls

set interfaces lo0 unit 0 family inet address 2.2.2.2/32

# IS-IS Configuration
set protocols isis level 2 wide-metrics-only
set protocols isis interface ge-1/0/0.0 level 2 passive
set protocols isis interface ge-2/0/0.0 level 2 passive

# BGP Configuration
set protocols bgp group IBGP type internal
set protocols bgp group IBGP local-address 2.2.2.2
set protocols bgp group IBGP peer-as 100
set protocols bgp group IBGP neighbor 1.1.1.1
set protocols bgp group IBGP neighbor 3.3.3.3

# Routing Policy
set policy-options policy-statement vpn1-import-policy term 1 from protocol bgp
set policy-options policy-statement vpn1-import-policy term 1 then accept


# OSPF Configuration
set protocols ospf area 0 interface lo0.0 passive
set protocols ospf area 0 interface ge-1/0/0.0
