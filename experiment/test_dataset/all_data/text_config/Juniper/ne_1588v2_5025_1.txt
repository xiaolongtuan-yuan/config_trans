set system host-name CE2

# ge-0/0/0接口PTP配置
set interfaces ge-0/0/0 unit 0 family inet address 192.168.0.2/24

# ge-0/0/1接口PTP配置
set interfaces ge-0/0/1 unit 0 family inet address 172.16.1.1/24

# 路由配置
set routing-options static route 192.168.0.0/24 next-hop 192.168.0.1
set routing-options static route 172.16.1.0/24 next-hop 172.16.1.2

# OSPF配置（添加逻辑单元号）
set protocols ospf area 0 interface ge-0/0/0.0
set protocols ospf area 0 interface ge-0/0/1.0
