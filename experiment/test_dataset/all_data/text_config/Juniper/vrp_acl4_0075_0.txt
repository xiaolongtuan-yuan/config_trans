set system host-name PE

# 路由实例配置
set routing-instances vpna instance-type virtual-router
set routing-instances vpna route-distinguisher 100:1
set routing-instances vpna interface ge-0/0/0.0

set routing-instances vpnb instance-type virtual-router
set routing-instances vpnb route-distinguisher 100:2
set routing-instances vpnb interface ge-0/0/1.0

# 路由策略配置
set policy-options community vpna-import members 111:1
set policy-options community vpna-export members 111:1
set policy-options community vpnb-import members 222:2
set policy-options community vpnb-export members 222:2

set policy-options policy-statement vpna-export-policy term 1 from protocol direct
set policy-options policy-statement vpna-export-policy term 1 then community add vpna-export
set policy-options policy-statement vpna-export-policy then accept

set policy-options policy-statement vpnb-export-policy term 1 from protocol direct
set policy-options policy-statement vpnb-export-policy then accept

# 接口配置
set interfaces ge-0/0/0 unit 0 family inet address 10.1.1.1/24
set interfaces ge-0/0/0 unit 0 family inet filter input acl-2001

set interfaces ge-0/0/1 unit 0 family inet address 10.2.1.1/24
set interfaces ge-0/0/1 unit 0 family inet filter input acl-2001

# 防火墙配置
set firewall family inet filter acl-2001 term 1 from source-address 10.1.1.0/24
set firewall family inet filter acl-2001 term 1 then accept

set firewall family inet filter acl-2001 term 2 from source-address 10.2.1.0/24
set firewall family inet filter acl-2001 term 2 then discard

# 用户管理配置
set system login user operator class super-user
set system login user operator authentication encrypted-password "$1c$`g=H/qo%;7$b6Za%2'[D!0blsOXF=.3QCNC-f)co,[aeE.`e`-<"
