hostname DS-Lite
#
license boot level ds-lite
#
ipv6 access-list 3500
 permit ipv6 2001:DB8:2::/64 any
#
ip vrf forwarding DS-Lite
 rd 1:1
ipv6 unicast-routing
!
interface GigabitEthernet2/0/2
 no shutdown

 ipv6 address 2001:DB8:1::2:1/64
 ipv6 traffic-policy p1 inbound
!

!
class-map match-all c1
 match access-group 3500
!
policy-map p1
 class c1
  set ip next-hop 2001:DB8:2::1
!
