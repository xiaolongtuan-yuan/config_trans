hostname Hub

interface GigabitEthernet1/0/0
 ip address 10.1.1.10 255.255.255.0
 tunnel source Loopback0
 tunnel mode gre ip
 tunnel key 100
 tunnel protection ipsec profile GRE-PROF

interface Loopback0
 ip address 192.168.0.1 255.255.255.255

interface Tunnel0
 ip address 172.16.1.1 255.255.255.0
 tunnel source GigabitEthernet1/0/0
 tunnel destination 172.16.1.2
 tunnel key 200
 tunnel protection ipsec profile GRE-PROF

router bgp 100
 bgp router-id 172.16.1.1
 neighbor 172.16.1.2 remote-as 200
 neighbor 172.16.1.3 remote-as 300
 !
 address-family ipv4
  no synchronization
  redistribute ospf 1
  neighbor 172.16.1.2 activate
  neighbor 172.16.1.3 activate
 exit-address-family

router ospf 1
 router-id 172.16.1.1
 network 192.168.0.0 0.0.0.255 area 0.0.0.0

router ospf 2
 router-id 10.1.1.10
 network 10.1.1.0 0.0.0.255 area 0.0.0.1
