set system host-name NATA

set routing-instances VRF1 instance-type vrf
set routing-instances VRF1 route-distinguisher 1:1
set routing-instances VRF1 vrf-target import target:1:1
set routing-instances VRF1 vrf-target export target:1:1
set routing-instances VRF1 interface ge-1/0/0.0

set firewall family inet filter ACL1 term match-traffic from source-address 192.168.10.0/24
set firewall family inet filter ACL1 term match-traffic then accept

set services nat pool NAT_POOL1 address-range low 11.11.11.101 high 11.11.11.105
set services nat pool NAT_POOL1 address-netmask 255.255.255.0
set services nat rule NAT_RULE1 match-direction input
set services nat rule NAT_RULE1 term 1 from source-address 192.168.10.0/24
set services nat rule NAT_RULE1 term 1 then translated-source pool NAT_POOL1

set interfaces ge-1/0/0 unit 0 family inet address 192.168.10.1/24
set interfaces ge-1/0/0 unit 0 family inet filter input ACL1
set interfaces ge-1/0/0 unit 0 vrf VRF1

set interfaces ge-1/0/1 unit 0 family inet address 1.1.1.1/24
set interfaces ge-1/0/1 disable
