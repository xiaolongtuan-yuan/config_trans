set system host-name NAT64_CGN
set system syslog host 1.1.1.1 port 514 source-address ge-0/0/1
set system syslog host 1.1.1.1 any info

set interfaces ge-0/0/1 unit 0 family inet6 address 2001:db8::1:110e/126
set security address-book global address IPV6_HOST_1 2001:db8::1:1112/126
set security policies from-zone trust to-zone nat64 policy IPV6_TO_NAT64 match source-address IPV6_HOST_1
set security policies from-zone trust to-zone nat64 policy IPV6_TO_NAT64 match destination-address NAT64_TRANSLATION
set security policies from-zone trust to-zone nat64 policy IPV6_TO_NAT64 then permit

set security nat64 rule-set NAT64-RULESET from zone trust
set security nat64 prefix 64:FF9B::/96
set security nat source pool NAT64_GROUP1 address 11.1.1.1/24
set security nat source rule-set NAT_SOURCE_RULE from zone trust
set security nat source rule-set NAT_SOURCE_RULE to zone nat64
set security nat source rule-set NAT_SOURCE_RULE rule NAT_SOURCE_RULE match source-address IPV6_HOST_1

set security nat destination pool NAT64_DEST_POOL address 11.1.1.1/24
set security nat destination rule-set NAT_DEST_RULE from zone nat64
set security nat destination rule-set NAT_DEST_RULE to zone trust

set applications application TCP_TIMEOUT term tcp protocol tcp
set applications application TCP_TIMEOUT term tcp inactivity-timeout 86400
set applications application UDP_TIMEOUT term udp protocol udp
set applications application UDP_TIMEOUT term udp inactivity-timeout 1200
set applications application ICMP_TIMEOUT term icmp protocol icmp
set applications application ICMP_TIMEOUT term icmp inactivity-timeout 10

set security nat64 timeout udp 1200
