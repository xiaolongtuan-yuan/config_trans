set system host-name CE1

set vlans VLAN10 vlan-id 10

set interfaces ge-1/0/1 unit 0 family ethernet-switching interface-mode access
set interfaces ge-1/0/1 unit 0 family ethernet-switching vlan members VLAN10
set interfaces ge-1/0/1 enable

set interfaces ge-1/0/2 unit 0 family ethernet-switching interface-mode trunk
set interfaces ge-1/0/2 unit 0 family ethernet-switching vlan members VLAN10
set interfaces ge-1/0/2 enable

set interfaces ge-1/0/2 native-vlan-id VLAN10  # 如果需要指定 Native VLAN（可选）
