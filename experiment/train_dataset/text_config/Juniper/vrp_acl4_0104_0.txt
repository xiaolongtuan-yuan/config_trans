set system host-name Router

set vlans vlan10 vlan-id 10
set vlans vlan20 vlan-id 20

set firewall family inet filter deny-ip term 1 from source-address 10.1.1.0/24
set firewall family inet filter deny-ip term 1 from destination-address 10.1.2.0/24
set firewall family inet filter deny-ip term 1 then discard

set firewall family inet filter deny-ip term 2 from source-address 10.1.2.0/24
set firewall family inet filter deny-ip term 2 from destination-address 10.1.1.0/24
set firewall family inet filter deny-ip term 2 then discard

set interfaces vlan unit 10 family inet address 10.1.1.1/24
set interfaces vlan unit 20 family inet address 10.1.2.1/24

set interfaces ge-0/0/0 unit 0 family ethernet-switching vlan members vlan10
set interfaces ge-0/0/0 unit 0 family ethernet-switching filter input deny-ip

set interfaces ge-0/0/1 unit 0 family ethernet-switching vlan members vlan20
set interfaces ge-0/0/1 unit 0 family ethernet-switching filter input deny-ip

set interfaces ge-0/0/0 unit 0 family ethernet-switching port-mode trunk
set interfaces ge-0/0/1 unit 0 family ethernet-switching port-mode trunk

set interfaces ge-0/0/0 unit 0 family ethernet-switching trunk allowed-vlans [10]
