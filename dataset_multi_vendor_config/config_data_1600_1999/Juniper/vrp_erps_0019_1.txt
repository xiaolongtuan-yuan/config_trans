# VLAN 配置

set vlans vlan20 vlan-id 20

set vlans vlan400 vlan-id 400

# STP 配置
set protocols spanning-tree mode rapid-pvst
set protocols spanning-tree extend-system-id
set protocols spanning-tree vlan 10 root primary
set protocols spanning-tree vlan 100-200 root primary
set protocols spanning-tree vlan 20 root primary
set protocols spanning-tree vlan 300-400 root primary

# ERPS 配置 - Ethernet Ring Protection Switching (ERPS)
set protocols erps ring 1 control-vlan 10
set protocols erps ring 1 protected-instance 1
set protocols erps ring 1 wtr-timer 6
set protocols erps ring 1 guard-timer 100

set protocols erps ring 2 control-vlan 20
set protocols erps ring 2 protected-instance 2
set protocols erps ring 2 wtr-timer 6
set protocols erps ring 2 guard-timer 100
