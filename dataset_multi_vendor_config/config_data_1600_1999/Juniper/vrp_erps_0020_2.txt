set system host-name DeviceC

# VLAN Configuration
set vlans vlan20 vlan-id 20
set vlans vlan300-400 vlan-id 300-400

# Spanning Tree Configuration
set protocols rstp region-configuration instance 2 vlan 20
set protocols rstp region-configuration instance 2 vlan 300-400
set protocols rstp instance 2 priority root primary

# ERPS Configuration
set protocols erps ring 2 control-vlan 20
set protocols erps ring 2 protected-instance 2
set protocols erps ring 2 wtr-timer 6
set protocols erps ring 2 guard-timer 100
set protocols erps ring 2 version 2
set protocols erps ring 2 sub-ring

# Interface Configuration
set interfaces ge-0/0/1 unit 0 family ethernet-switching port-mode trunk
set interfaces ge-0/0/1 unit 0 family ethernet-switching trunk allowed-vlans 20,300-400
set interfaces ge-0/0/1 unit 0 family ethernet-switching spanning-tree portfast
set protocols erps ring 2 owner ge-0/0/1

set interfaces ge-0/0/2 unit 0 family ethernet-switching port-mode trunk
set interfaces ge-0/0/2 unit 0 family ethernet-switching trunk allowed-vlans 20,300-400
set interfaces ge-0/0/2 unit 0 family ethernet-switching spanning-tree portfast
set protocols erps ring 2 ge-0/0/2
