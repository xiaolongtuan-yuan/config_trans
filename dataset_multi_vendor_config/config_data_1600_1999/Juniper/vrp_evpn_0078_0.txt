set system host-name PE1

set protocols mpls label-protocol ldp
set protocols mpls ldp router-id lo0 force

set interfaces et-0/0/10 gigether-options 802.3ad ae10
set interfaces et-0/0/10.1 unit 0 vlan-id 100 family ethernet-switching bridge-domain 10

set interfaces ge-0/0/0 unit 0 family inet address 10.1.1.1/24
set interfaces ge-0/0/0 unit 0 family mpls

set interfaces lo0 unit 0 family inet address 1.1.1.1/32

set protocols bgp group external type external
set protocols bgp group external local-address 1.1.1.1
set protocols bgp group external neighbor 3.3.3.3 peer-as 100
set protocols bgp group external neighbor 3.3.3.3 update-source lo0

set protocols bgp group external family inet unicast
set protocols bgp group external family vpnv4 unicast

set protocols ospf area 0.0.0.0 interface lo0
set protocols ospf area 0.0.0.0 interface ge-0/0/0

set protocols evpn source-interface lo0

set ethernet-switching-options bridge-domains 10 vlan-id 100
set ethernet-switching-options bridge-domains 10 routing-interface vlan.10

set vlans vlan100 vlan-id 100

# EVPN configuration
set routing-instances evrf1 instance-type virtual-router
set routing-instances evrf1 route-distinguisher 100:1
set routing-instances evrf1 vrf-import vpn-target
set routing-instances evrf1 vrf-export vpn-target
set routing-instances evrf1 bridge-domains 10

set protocols evpn vpn-instance evrf1
set protocols evpn mldp p2mp root-ip 1.1.1.1

# MPLS configuration
set protocols mpls lsr-id 1.1.1.1
set protocols mpls mldp p2mp

# Interface configurations
set interfaces et-0/0/10.1 encapsulation vlan-encap
set interfaces et-0/0/10.1 ether-options 802.3ad ae10
set interfaces ge-0/0/0 ether-options 802.3ad ae10

# Enable MPLS and LDP
set protocols mpls interface ge-0/0/0
set protocols mpls ldp

# BGP configuration
set protocols bgp group external family inet unicast
set protocols bgp group external family
