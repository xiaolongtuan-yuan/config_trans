set interfaces ge-0/0/0 unit 0 mpls
set interfaces ge-0/0/0.10 unit 0 vlan-id 10 family ethernet-switching
set interfaces ge-0/0/0.10 unit 0 encapsulation vlan-bridge

# ISIS Configuration
set protocols isis level 2 net 49.0001.0001.0001.0001.00

set protocols isis segment-routing mpls

# BGP Configuration
set protocols bgp group external neighbor 1.1.1.1 remote-as 100
set protocols bgp group external neighbor 1.1.1.1 update-source lo0

set protocols bgp group external family l2vpn evpn activate

set protocols evpn vpn-instance evrf1 vpn-target 1:1 import-extcommunity
