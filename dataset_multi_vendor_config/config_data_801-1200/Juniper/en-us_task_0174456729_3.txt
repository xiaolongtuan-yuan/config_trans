set system host-name DeviceD

# MPLS和LDP配置
set protocols ldp interface all
set protocols ldp router-id lo0.0

# IS-IS路由器ID和基本配置
set protocols isis router-id 4.4.4.9
set protocols isis level 1 wide-metrics-only
set protocols isis interface ge-1/0/0 level 1
set protocols isis interface ge-1/0/0 point-to-point
set protocols isis interface ge-1/0/0 level 1 metric 100
set protocols isis interface ge-2/0/0 level 1
set protocols isis interface ge-2/0/0 point-to-point
set protocols isis interface lo0.1 passive
set protocols isis interface lo0.1 point-to-point
set protocols isis lo0.1 address-family inet unicast prefix-sid index 40

# 设置 NET
set protocols isis net 10.0000.0000.0004.00

# MPLS TE 配置
set protocols isis traffic-engineering te-router-id lo0.0
set protocols isis traffic-engineering area 0

# 接口配置 - GigabitEthernet1/0/0 对应于 ge-1/0/0
set interfaces ge-1/0/0 unit 0 family inet address 10.3.1.2/24

# 接口配置 - GigabitEthernet2/0/0 对应于 ge-2/0/0
set interfaces ge-2/0/0 unit 0 family inet address 10.4.1.1/24

# 接口配置 - Loopback1 对应于 lo0.1
set interfaces lo0 unit 1 family inet address 4.4.4.9/32
