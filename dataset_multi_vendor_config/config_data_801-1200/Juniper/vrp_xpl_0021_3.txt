# 设置设备名称
set system host-name DeviceD

# 配置接口 ge-3/0/1
set interfaces ge-3/0/1 unit 0 family inet address 1.1.2.2/24
set interfaces ge-3/0/1 unit 0 enable

# 配置接口 ge-3/0/2
set interfaces ge-3/0/2 unit 0 family inet address 1.1.4.1/24
set interfaces ge-3/0/2 unit 0 enable

# 配置BGP协议
set protocols bgp group external-peers type external
set protocols bgp group external-peers local-as 200

# 与邻居1.1.2.1的配置
set protocols bgp group external-peers neighbor 1.1.2.1 peer-as 200
set protocols bgp group external-peers neighbor 1.1.2.1 family inet unicast

# 与邻居1.1.4.2的配置
set protocols bgp group external-peers neighbor 1.1.4.2 peer-as 100
set protocols bgp group external-peers neighbor 1.1.4.2 family inet unicast

# 关闭同步，Juniper中BGP默认没有同步设置，无需配置
