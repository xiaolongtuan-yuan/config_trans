# 设置主机名
set system host-name DeviceF

# 配置IS-IS路由协议
set protocols isis level 2 wide-metrics-only 
set protocols isis level 2 lsp-triggers immediate
set protocols isis level 2 multilevel-hello-interval 1
set protocols isis interface ge-1/0/0.0 level 2
set protocols isis interface ge-2/0/0.0 level 2
set protocols isis interface all passive
deactivate protocols isis interface all
set protocols isis no-narrow-metrics
set protocols isis level 1 disable
set protocols isis multimode
set protocols isis level 2 no-overload-sos active

# 配置NET地址
set protocols isis level 2 net 49.0006.6666.6666.0000.00

# 配置接口GigabitEthernet1/0/0
set interfaces ge-1/0/0 unit 0 family inet address 192.168.2.5/24
set interfaces ge-1/0/0 unit 0 family iso
set protocols isis interface ge-1/0/0.0 point-to-point
set protocols isis interface ge-1/0/0 level 2

# 配置接口GigabitEthernet2/0/0
set interfaces ge-2/0/0 unit 0 family inet address 192.168.0.3/24
set interfaces ge-2/0/0 unit 0 family iso
set protocols isis interface ge-2/0/0.0 point-to-point
set protocols isis interface ge-2/0/0 level 2
