# 设置主机名
set system host-name CE1

# 配置物理接口
set interfaces ge-1/0/0 unit 0 family inet address 10.1.1.2/24
set interfaces ge-1/0/0 disable

# 配置逻辑接口
set interfaces lo0 unit 1 family inet address 11.11.11.11/32

# 配置BGP
set protocols bgp group external-peers type external
set protocols bgp group external-peers neighbor 10.1.1.1 peer-as 100
set protocols bgp group external-peers family inet unicast
set protocols bgp group external-peers family inet no-synchronization
set protocols bgp group external-peers export connected-routes
set policy-options policy-statement connected-routes term 1 from protocol direct
set policy-options policy-statement connected-routes term 1 then accept

# 配置自有网络广告
set policy-options policy-statement advertise-lo0 from interface lo0.1
set policy-options policy-statement advertise-lo0 then accept
set protocols bgp group external-peers export advertise-lo0
