set system host-name Router

set vlans VLAN10 vlan-id 10
set vlans VLAN20 vlan-id 20

set firewall family inet filter acl-3001 term deny-ip from source-address 10.1.1.0/24
set firewall family inet filter acl-3001 term deny-ip from destination-address 10.1.2.0/24
set firewall family inet filter acl-3001 term deny-ip then discard

set firewall family inet filter acl-3002 term deny-ip from source-address 10.1.2.0/24
set firewall family inet filter acl-3002 term deny-ip from destination-address 10.1.1.0/24
set firewall family inet filter acl-3002 term deny-ip then discard

set firewall family inet filter cm1 term match-acl-3001 filter acl-3001
set firewall family inet filter cm1 term match-acl-3002 filter acl-3002
set firewall family inet filter cm1 term match-acl-3002 then discard

set interfaces vlan unit 10 family inet address 10.1.1.1/24
set interfaces vlan unit 20 family inet address 10.1.2.1/24

set interfaces ge-1/0/0 unit 0 family ethernet-switching port-mode trunk
set interfaces ge-1/0/0 unit 0 family ethernet-switching vlan members VLAN10
set interfaces ge-1/0/0 unit 0 family inet filter input cm1

set interfaces ge-2/0/0 unit 0 family ethernet-switching port-mode trunk
set interfaces ge-2/0/0 unit 0 family ethernet-switching vlan members VLAN20
set interfaces ge-2/0/0 unit 0 family inet filter input cm1
