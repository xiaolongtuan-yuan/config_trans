set system host-name MCE

# VRF configurations
set routing-instances vpna instance-type vrf
set routing-instances vpna route-distinguisher 100:1

set routing-instances vpnb instance-type vrf
set routing-instances vpnb route-distinguisher 200:2

# Interface configurations
set interfaces ge-1/0/0 unit 0 family inet address 10.5.1.2/24
set routing-instances vpna interface ge-1/0/0.0

set interfaces ge-2/0/0 unit 0 family inet address 10.5.2.2/24
set routing-instances vpnb interface ge-2/0/0.0

set interfaces ge-3/0/0 unit 0 family inet address 10.3.1.2/24
set routing-instances vpna interface ge-3/0/0.0

set interfaces ge-4/0/0 unit 0 family inet address 10.4.1.2/24
set routing-instances vpnb interface ge-4/0/0.0

# OSPF configurations
set protocols ospf area 0.0.0.0 network 10.3.1.0/24
set protocols ospf area 0.0.0.0 network 10.5.1.0/24
set protocols ospf export redis-rip
set routing-instances vpna protocols ospf import redis-ospf

set protocols ospf area 0.0.0.0 network 10.4.1.0/24
set protocols ospf area 0.0.0.0 network 10.5.2.0/24
set routing-instances vpnb protocols ospf import redis-ospf

# RIP configurations
set protocols rip group GROUP source-address 10.3.1.2
set protocols rip group GROUP export redis-ospf
set routing-instances vpna protocols rip import redis-rip

set protocols rip group GROUP source-address 10.4.1.2
set protocols rip group GROUP export redis-ospf
set routing-instances vpnb protocols rip import redis-rip

# Policies for redistribution
set policy-options policy-statement redis-ospf term 1 from protocol ospf
set policy-options policy-statement redis-ospf term 1 then accept

set policy-options policy-statement redis-rip term 1 from protocol rip
set policy-options policy-statement redis-rip term 1 then accept
