hostname MCE


vrf definition vpna
 rd 100:1
!


vrf definition vpnb
 rd 200:2
!


interface GigabitEthernet1/0/0
 no shutdown
 vrf forwarding vpna
 ip address 10.5.1.2 255.255.255.0
!


interface GigabitEthernet2/0/0
 no shutdown
 vrf forwarding vpnb
 ip address 10.5.2.2 255.255.255.0
!


interface GigabitEthernet3/0/0
 no shutdown
 vrf forwarding vpna
 ip address 10.3.1.2 255.255.255.0
!


interface GigabitEthernet4/0/0
 no shutdown
 vrf forwarding vpnb
 ip address 10.4.1.2 255.255.255.0
!


router ospf 100 vrf vpna
 redistribute rip subnets
 capability vrf-lite
 network 10.3.1.0 0.0.0.255 area 0
 network 10.5.1.0 0.0.0.255 area 0
!


router ospf 200 vrf vpnb
 redistribute rip subnets
 capability vrf-lite
 network 10.4.1.0 0.0.0.255 area 0
 network 10.5.2.0 0.0.0.255 area 0
!


router rip vrf vpna
 version 2
 network 10.0.0.0
 redistribute ospf 100 metric 1
!


router rip vrf vpnb
 version 2
 network 10.0.0.0
 redistribute ospf 200 metric 1
!
