hostname RR1
!
mpls label protocol ldp
!
interface GigabitEthernet1/0/0
 no shutdown
 ip address 10.10.2.2 255.255.255.0
 mpls ip
 mpls ldp autoconfig
!
interface GigabitEthernet2/0/0
 no shutdown
 ip address 10.20.3.1 255.255.255.0
 mpls ip
 mpls ldp autoconfig
!
interface GigabitEthernet3/0/0
 no shutdown
 ip address 10.20.4.1 255.255.255.0
 mpls ip
 mpls ldp autoconfig
!
interface Loopback1
 ip address 2.2.2.9 255.255.255.255
!
router bgp 100
 neighbor 1.1.1.9 remote-as 100
 neighbor 1.1.1.9 update-source Loopback1
 neighbor 3.3.3.9 remote-as 100
 neighbor 3.3.3.9 update-source Loopback1
 neighbor 4.4.4.9 remote-as 100
 neighbor 4.4.4.9 update-source Loopback1
 !
 address-family ipv4 unicast
  neighbor 4.4.4.9 activate
  neighbor 1.1.1.9 activate
  neighbor 3.3.3.9 activate
  no synchronization
 exit-address-family
 !
 address-family vpnv4
  bgp additional-paths select best external
  neighbor 1.1.1.9 activate
  neighbor 1.1.1.9 route-reflector-client
  neighbor 3.3.3.9 activate
  neighbor 3.3.3.9 route-reflector-client
  neighbor 4.4.4.9 activate
  neighbor 4.4.4.9 route-reflector-client
  exit-address-family
!
router ospf 1
 network 10.10.2.0 0.0.0.255 area 0
 network 10.20.3.0 0.0.0.255 area 0
 network 10.20.4.0 0.0.0.255 area 0
 network 2.2.2.9 0.0.0.0 area 0
!
