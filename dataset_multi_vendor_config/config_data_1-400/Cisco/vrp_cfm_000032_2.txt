hostname PE1

mpls label protocol ldp

mpls ldp router-id Loopback1 force

interface GigabitEthernet1/0/0.10
  no shutdown
  encapsulation dot1q 10
  xconnect vfi bgp1

interface GigabitEthernet2/0/0
  no shutdown
  ip address 192.168.1.1 255.255.255.0
  mpls ip

interface Loopback1
  ip address 1.1.1.9 255.255.255.255

router bgp 100
  neighbor 3.3.3.9 remote-as 100
  neighbor 3.3.3.9 update-source Loopback1
  address-family l2vpn vpls
    neighbor 3.3.3.9 activate
    neighbor 3.3.3.9 send-community extended
    signaling protocol ldp

router ospf 1
  network 1.1.1.9 0.0.0.0 area 0
  network 192.168.0.0 0.0.0.255 area 0

l2 vfi bgp1 manual
  vpn id 1
  autodiscovery bgp signaling bgp
  autodiscovery bgp route-target
    import 100:1
    export 100:1
  bridge-domain 5

ethernet cfm ma ma1
  mep domain bgp1
    mep 1
      interface GigabitEthernet1/0/0.10 direction inward
      ccm
    mep 2
      remote-mep

end
