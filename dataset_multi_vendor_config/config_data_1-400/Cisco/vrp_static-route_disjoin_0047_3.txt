hostname DeviceC

bfd all-interfaces

interface GigabitEthernet1/0/2
 no shutdown
 ipv6 enable
 ipv6 address 2001:db8:6::1/64

interface GigabitEthernet2/0/3
 no shutdown
 ipv6 enable
 ipv6 address 2001:db8:3::2/64

 no shutdown
 ipv6 enable
 ipv6 address 2001:db8:4::2/64
