hostname ASBR2

mpls ldp router-id Loopback0
!

 no shutdown

 mpls ldp auto-config
!
interface GigabitEthernet1/0/4
 no shutdown
 ip address 10.0.16.1 255.255.255.0

 mpls ldp auto-config
!
interface Loopback0
 ip address 1.1.1.4 255.255.255.255
!

 is-type level-2-only
 metric-style wide

 neighbor 1.1.1.5 remote-as 200
 neighbor 1.1.1.5 update-source Loopback0
 !

  neighbor 10.0.7.5 activate
  neighbor 10.0.7.5 route-policy policy1 export
  neighbor 10.0.7.5 next-hop-self
  neighbor 1.1.1.5 activate
  neighbor 1.1.1.5 route-policy policy2 export
  neighbor 1.1.1.5 next-hop-self
  no synchronization
  network 1.1.1.5 mask 255.255.255.255

!

  if-match mpls-label
  then label allocate 1

  then label allocate 1
