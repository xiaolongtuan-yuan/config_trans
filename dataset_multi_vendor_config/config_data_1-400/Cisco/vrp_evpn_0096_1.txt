hostname ASBR1

mpls label protocol ldp

interface GigabitEthernet1/0/0
 no shutdown
 ip address 10.1.1.1 255.255.255.0
 mpls ip

interface GigabitEthernet2/0/0
 no shutdown
 ip address 10.2.1.1 255.255.255.0
 mpls ip

interface Loopback1
 ip address 2.2.2.2 255.255.255.255

router bgp 100
 neighbor 1.1.1.1 remote-as 100
 neighbor 1.1.1.1 update-source Loopback1
 neighbor 10.2.1.2 remote-as 200

 address-family ipv4 unicast
  no synchronization
  network 1.1.1.1 mask 255.255.255.255
  network 10.1.1.0 mask 255.255.255.0
  neighbor 1.1.1.1 activate
  neighbor 1.1.1.1 route-map policy2 out
  neighbor 1.1.1.1 send-label
  neighbor 10.2.1.2 activate
  neighbor 10.2.1.2 route-map policy1 out
  neighbor 10.2.1.2 send-label

router ospf 1
 network 2.2.2.2 0.0.0.0 area 0
 network 10.1.1.0 0.0.0.255 area 0

route-map policy1 permit 10
 match mpls-label
 set mpls-label

route-map policy2 permit 10
 match mpls-label
 set mpls-label

end
