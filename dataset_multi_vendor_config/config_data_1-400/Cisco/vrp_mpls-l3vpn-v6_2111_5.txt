hostname CE2
!
mpls label protocol ldp
!
interface GigabitEthernet1/0/0
 no shutdown
 ip address 10.5.1.2 255.255.255.0
 mpls ip
!
interface GigabitEthernet2/0/0
 no shutdown
 ip address 10.2.1.1 255.255.255.0
 mpls ip
!
interface Loopback1
 ip address 5.5.5.9 255.255.255.255
!
router bgp 300
 neighbor 10.5.1.1 remote-as 100
 neighbor 6.6.6.9 remote-as 300
 neighbor 6.6.6.9 update-source Loopback1
 !
 address-family ipv4
  no synchronization
  network 5.5.5.9 mask 255.255.255.255
  network 10.2.1.0 mask 255.255.255.0
  network 10.5.1.0 mask 255.255.255.0
  neighbor 10.5.1.1 activate
  neighbor 10.5.1.1 route-map policy1 out
  neighbor 10.5.1.1 send-label
  neighbor 6.6.6.9 activate
  neighbor 6.6.6.9 route-map policy2 out
  neighbor 6.6.6.9 send-label
 exit-address-family
!
route-map policy1 permit 10
 set mpls label
!
route-map policy2 permit 10
 match mpls label
 set mpls label
!
router ospf 1
 network 5.5.5.9 0.0.0.0 area 0
 network 10.2.1.0 0.0.0.255 area 0
 network 10.5.1.0 0.0.0.255 area 0
!
end
