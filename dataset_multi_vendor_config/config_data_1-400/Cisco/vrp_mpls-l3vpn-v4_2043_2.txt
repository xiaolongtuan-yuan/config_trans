hostname CE1

mpls ip

interface GigabitEthernet1/0/0
 no shutdown
 ip address 40.1.1.2 255.255.255.0
 mpls ip

interface GigabitEthernet2/0/0
 no shutdown
 ip address 11.1.1.1 255.255.255.0
 mpls ip

interface Loopback1
 ip address 2.2.2.9 255.255.255.255

router bgp 200
 neighbor 11.1.1.2 remote-as 100
 neighbor 1.1.1.9 remote-as 200
 neighbor 1.1.1.9 update-source Loopback1
 !
 address-family ipv4
  no synchronization
  redistribute ospf 1
  neighbor 11.1.1.2 activate
  neighbor 11.1.1.2 route-map policy1 out
  neighbor 11.1.1.2 send-label
  neighbor 1.1.1.9 activate
  neighbor 1.1.1.9 route-map policy2 out
  neighbor 1.1.1.9 send-label
 exit-address-family

router ospf 1
 network 2.2.2.9 0.0.0.0 area 0
 network 40.1.1.0 0.0.0.255 area 0
 network 11.1.1.0 0.0.0.255 area 0

route-map policy1 permit 10
 set mpls label

route-map policy2 permit 10
 match mpls label
 set mpls label

end
