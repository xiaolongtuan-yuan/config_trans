hostname DeviceB

bfd all-interfaces

interface GigabitEthernet1/0/0
 no shutdown
 ip address 10.1.1.2 255.255.255.0
end

interface GigabitEthernet2/0/0
 no shutdown
 ip address 10.3.1.1 255.255.255.0
end

router bgp 200
 bgp router-id 2.2.2.2
 neighbor 10.1.1.1 remote-as 100
 neighbor 10.1.1.1 bfd interval 100 min_rx 100 multiplier 4
 neighbor 10.3.1.2 remote-as 200
 neighbor 10.3.1.2 bfd interval 100 min_rx 100 multiplier 4
end

address-family ipv4
 no synchronization
 neighbor 10.1.1.1 route-map rtb export
 neighbor 10.1.1.1 activate
 neighbor 10.3.1.2 activate
exit-address-family

route-map rtb permit 10
 set metric 80
end
