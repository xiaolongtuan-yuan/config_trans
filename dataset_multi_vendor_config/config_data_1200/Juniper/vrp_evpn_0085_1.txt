# Set hostname
set system host-name SPE

# Configure routing-instance (VRF)
set routing-instances vpn1 instance-type vrf
set routing-instances vpn1 route-distinguisher 100:1
set routing-instances vpn1 vrf-target target:2:2

# Configure MPLS for LDP
set protocols mpls label-switched-path ldp-signaling
set protocols ldp interface all
set protocols ldp router-id 2.2.2.2 force

# Interface configurations
set interfaces ge-1/0/0 unit 0 family inet address 10.1.1.2/24
set interfaces ge-1/0/0 unit 0 family mpls
set interfaces ge-2/0/0 unit 0 family inet address 10.2.1.1/24
set interfaces ge-2/0/0 unit 0 family mpls

# Loopback interface
set interfaces lo0 unit 1 family inet address 2.2.2.2/32

# BGP Configuration
set protocols bgp group internal type internal
set protocols bgp group internal local-address 2.2.2.2
set protocols bgp group internal neighbor 1.1.1.1 peer-as 100
set protocols bgp group internal neighbor 1.1.1.1 family inet unicast
set protocols bgp group internal neighbor 1.1.1.1 family inet-vpn unicast
set protocols bgp group internal neighbor 3.3.3.3 peer-as 100
set protocols bgp group internal neighbor 3.3.3.3 family inet unicast
set protocols bgp group internal neighbor 3.3.3.3 family inet-vpn unicast

# Export default route
set routing-options static route 0.0.0.0/0 next-hop discard
set policy-options policy-statement default-route
set policy-options policy-statement default-route term 1 from route-filter 0.0.0.0/0 exact
set policy-options policy-statement default-route term 1 then accept

# OSPF Configuration
set protocols ospf area 0 interface lo0.1
set protocols ospf area 0 interface ge-1/0/0.0

# L2VPN Configuration
set protocols l2vpn l2vpn-name vpn1 description "L2VPN between interfaces"
set protocols l2vpn l2vpn-name vpn1 interface ge-1/0/0.0
set protocols l2vpn l2vpn-name vpn1 interface ge-2/0/0.0

# Static route in VRF
set routing-instances vpn1 routing-options static route 0.0.0.0/0 next-hop discard
