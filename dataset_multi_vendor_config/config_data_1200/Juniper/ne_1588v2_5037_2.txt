set system host-name DeviceC

# Loopback Interface
set interfaces lo0 unit 0 family inet address 3.3.3.3/32

# NTP Configuration
set system ntp server 1.1.1.1
set system ntp server 2.2.2.2
set system ntp source-address 3.3.3.3
set system ntp authentication-key 1 type md5
set system ntp master 5

# Clock Configuration
set system ptp clock-source internal
set system ptp clock-source priority 1
set system ptp clock-source ssm prc
