# 设置设备主机名
set system host-name CE2

# 启用多播转发
set protocols pim interface all mode sparse

# 配置接口GigabitEthernet1/0/0
set interfaces ge-1/0/0 unit 0 family inet address 192.168.2.2/24
set protocols pim interface ge-1/0/0 mode sparse

# 配置接口GigabitEthernet1/0/1
set interfaces ge-1/0/1 unit 0 family inet address 10.1.4.1/24
set protocols pim interface ge-1/0/1 mode sparse
set protocols igmp interface ge-1/0/1 version 3

# 配置Loopback1接口
set interfaces lo0 unit 1 family inet address 5.5.5.5/32

# 配置OSPF
set protocols ospf area 0.0.0.0 interface lo0.1
set protocols ospf area 0.0.0.0 interface ge-1/0/1
set protocols ospf area 0.0.0.0 interface ge-1/0/0

# 配置PIM RP地址
set protocols pim rp static address 1.1.1.1

# 确保所有接口开启
delete interfaces ge-1/0/0 disable
delete interfaces ge-1/0/1 disable
delete interfaces lo0 disable
