# 配置主机名称
set system host-name PE3

# 配置MPLS LDP和标签分配协议
set protocols ldp interface all-mpls-interfaces
set protocols ldp interface lo0.0

# 配置VPLS实例
set routing-instances VSI1 instance-type vpls
set routing-instances VSI1 route-distinguisher 192.168.0.0:1
set routing-instances VSI1 vrf-target target:100:1
set routing-instances VSI1 protocols vpls site local-site site-identifier 1
set routing-instances VSI1 protocols vpls mac-withdraw
set routing-instances VSI1 protocols vpls mac-withdraw propagate
set routing-instances VSI1 protocols vpls vpls-id 1
set routing-instances VSI1 protocols vpls neighbor 1.1.1.1

# 配置接口
set interfaces ge-1/0/0 unit 0 family inet address 192.168.5.1/24
set interfaces ge-1/0/1 unit 0 family inet address 192.168.2.2/24
set interfaces ge-1/0/2 unit 0 family inet address 192.168.3.1/24
set interfaces lo0 unit 0 family inet address 3.3.3.3/32

# 启用MPLS并进行LDP自动配置
set interfaces ge-1/0/0 unit 0 family mpls
set protocols ldp interface ge-1/0/0.0
set interfaces ge-1/0/1 unit 0 family mpls
set protocols ldp interface ge-1/0/1.0
set interfaces ge-1/0/2 unit 0 family mpls
set protocols ldp interface ge-1/0/2.0

# 配置BGP
set protocols bgp local-as 100
set protocols bgp group bgp-peers type internal
set protocols bgp group bgp-peers neighbor 2.2.2.2 peer-as 100
set protocols bgp group bgp-peers neighbor 2.2.2.2 local-address 3.3.3.3
set protocols bgp group bgp-peers neighbor 4.4.4.4 peer-as 100
set protocols bgp group bgp-peers neighbor 4.4.4.4 local-address 3.3.3.3
set protocols bgp group bgp-peers family inet-vpn any
set protocols bgp group bgp-peer family vpls-vpn any

# 配置OSPF
set protocols ospf area 0.0.0.0 interface lo0.0
set protocols ospf area 0.0.0.0 interface ge-1/0/0.0
set protocols ospf area 0.0.0.0 interface ge-1/0/1.0
set protocols ospf area 0.0.0.0 interface ge-1/0/2.0
set protocols ospf router-id 3.3.3.3
