set system host-name PE1

set vlans VLAN10 vlan-id 10
set vlans VLAN100 vlan-id 100

set protocols rstp bridge-priority 32768
set protocols rstp vlan VLAN10 instance 0
set protocols rstp vlan VLAN100 instance 0

set protocols erps ring 1 control-vlan VLAN100
set protocols erps ring 1 version v2
set protocols erps ring 1 sub-ring

set protocols mpls lsr-id 1.1.1.1
set protocols mpls interface all

set protocols l2vpn
set routing-instances s1 instance-type l2vpn
set routing-instances s1 interface vlan.10
set routing-instances s1 l2vpn encapsulation-type ethernet
set routing-instances s1 l2vpn signaling protocol ldp
set routing-instances s1 l2vpn vsi-id 10
set routing-instances s1 l2vpn neighbors 2.2.2.2
set routing-instances s1 l2vpn neighbors 3.3.3.3

set interfaces vlan unit 10 family ethernet-switching vlan members VLAN10
set interfaces vlan unit 10 family mpls

set interfaces ge-1/0/0 description "TRUNK PORT"
set interfaces ge-1/0/0 unit 0 family ethernet-switching port-mode trunk
set interfaces ge-1/0/0 unit 0 family ethernet-switching vlan members [ VLAN10 VLAN100 ]
set protocols stp disable
set protocols erps ring 1 interface ge-1/0/0

set interfaces ge-2/0/0 unit 0 family inet address 10.1.1.1/24
set interfaces ge-2/0/0 unit 0 family mpls
set protocols ldp interface ge-2/0/0.0

set interfaces ge-3/0/0 unit 0 family inet address 10.3.1.1/24
set interfaces ge-3/0/0 unit 0 family mpls
set protocols ldp interface ge-3/0/0.0

set interfaces lo0 unit 1 family inet address 1.1.1.1/32

set protocols ospf area 0.0.0.0 interface lo0.1
set protocols ospf area 0.0.0.0 interface ge-2/0/0.0
set protocols ospf area 0.0.0.0 interface ge-3/0/0.0
