set system host-name DeviceA

set vlans vlan10 vlan-id 10
set vlans vlan20 vlan-id 20
set vlans vlan100-200 vlan-id-range 100-200
set vlans vlan300-400 vlan-id-range 300-400

set protocols mstp instance 1 vlans [ vlan10 vlan100-200 ]
set protocols mstp instance 2 vlans [ vlan20 vlan300-400 ]

set protocols erps ring 1 control-vlan 10
set protocols erps ring 1 protected-instance 1
set protocols erps ring 1 wtr-timer 6
set protocols erps ring 1 guard-timer 100

set protocols erps ring 2 control-vlan 20
set protocols erps ring 2 protected-instance 2
set protocols erps ring 2 wtr-timer 6
set protocols erps ring 2 guard-timer 100

set interfaces ge-1/0/1 unit 0 family ethernet-switching interface-mode trunk
set interfaces ge-1/0/1 unit 0 family ethernet-switching vlan members [ vlan10 vlan20 vlan100-200 vlan300-400 ]
set interfaces ge-1/0/1 unit 0 disable
set protocols mstp interface ge-1/0/1 disable
set protocols erps ring 1 interface ge-1/0/1
set protocols erps ring 2 interface ge-1/0/1 rpl-owner

set interfaces ge-1/0/2 unit 0 family ethernet-switching interface-mode trunk
set interfaces ge-1/0/2 unit 0 family ethernet-switching vlan members [ vlan10 vlan20 vlan100-200 vlan300-400 ]
set interfaces ge-1/0/2 unit 0 disable
set protocols mstp interface ge-1/0/2 disable
set protocols erps ring 1 interface ge-1/0/2
set protocols erps ring 2 interface ge-1/0/2
