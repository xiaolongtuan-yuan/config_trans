set system host-name DeviceB

set interfaces ge-1/0/0 unit 0 family inet address 10.1.3.1/24
set interfaces ge-1/0/0 unit 0 description "GigabitEthernet1/0/0"

set interfaces ge-2/0/0 unit 0 family inet address 10.1.2.2/24
set interfaces ge-2/0/0 unit 0 description "GigabitEthernet2/0/0"

set routing-options autonomous-system 20
set routing-options router-id 2.2.2.2

set protocols bgp group AS10 type external
set protocols bgp group AS10 peer-as 10
set protocols bgp group AS10 neighbor 10.1.2.1

set protocols bgp group AS30 type external
set protocols bgp group AS30 peer-as 30
set protocols bgp group AS30 neighbor 10.1.3.2

set policy-options policy-statement FILTER_AS10 term 1 from as-path AS30
set policy-options policy-statement FILTER_AS10 term 1 then reject
set policy-options policy-statement FILTER_AS10 term 2 then accept

set policy-options policy-statement FILTER_AS30 term 1 from as-path AS10
set policy-options policy-statement FILTER_AS30 term 1 then reject
set policy-options policy-statement FILTER_AS30 term 2 then accept

set protocols bgp group AS10 export FILTER_AS10
set protocols bgp group AS30 export FILTER_AS30

set routing-options static route 10.1.3.0/24 discard
set routing-options static route 10.1.2.0/24 discard
