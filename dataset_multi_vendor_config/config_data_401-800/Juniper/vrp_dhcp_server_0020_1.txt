set system host-name "DHCP Server"

set system services dhcp-local-server group pool1
set system services dhcp-local-server group pool1 interface ge-1/0/1.1
set system services dhcp-local-server group pool1 subnet 10.10.10.0/24
set system services dhcp-local-server group pool1 default-router 10.10.10.1
set system services dhcp-local-server group pool1 dns-server <DNS_SERVER_IP>

set system services dhcp-local-server group pool2
set system services dhcp-local-server group pool2 interface ge-1/0/1.2
set system services dhcp-local-server group pool2 subnet 10.10.20.0/24
set system services dhcp-local-server group pool2 default-router 10.10.20.1
set system services dhcp-local-server group pool2 dns-server <DNS_SERVER_IP>

set interfaces ge-1/0/1 unit 0 family inet address 10.1.1.1/24

set interfaces ge-1/0/1 unit 1 vlan-id 100
set interfaces ge-1/0/1 unit 1 family inet address 10.10.10.1/24

set interfaces ge-1/0/1 unit 2 vlan-id 200
set interfaces ge-1/0/1 unit 2 family inet address 10.10.20.1/24
