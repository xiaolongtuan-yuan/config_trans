set protocols rstp vlan 10

set protocols erps instance 1 control-vlan 100
set protocols erps instance 1 protected-instance 0
set protocols erps instance 1 version 2
set protocols erps instance 1 sub-ring

set protocols mpls interface lo0.0
set protocols mpls interface ge-2/0/0.0
set protocols mpls interface ge-3/0/0.0

set protocols ldp interface lo0.0
set protocols ldp interface ge-2/0/0.0
set protocols ldp interface ge-3/0/0.0

set protocols l2vpn

set protocols l2vpn pseudowire-class s1 encapsulation ldp

set protocols l2vpn vfi s1 vpn-id 10
set protocols l2vpn vfi s1 neighbor 1.1.1.1 pw-id 10
set protocols l2vpn vfi s1 neighbor 3.3.3.3 pw-id 10
