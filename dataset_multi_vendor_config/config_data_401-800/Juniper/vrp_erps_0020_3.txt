set system host-name DeviceD

set vlans VLAN10 vlan-id 10
set vlans VLAN20 vlan-id 20
set vlans VLAN100-200 vlan-id 100-200
set vlans VLAN300-400 vlan-id 300-400

set protocols rstp vlan VLAN10 root-priority 0
set protocols rstp vlan VLAN100-200 root-priority 0
set protocols rstp vlan VLAN20 root-priority 0
set protocols rstp vlan VLAN300-400 root-priority 0

set protocols erps ring 1 control-vlan 10
set protocols erps ring 1 protected-instance 1
set protocols erps ring 1 version 2
set protocols erps ring 1 wtr-timer 6
set protocols erps ring 1 guard-timer 100
set protocols erps ring 1 tc-notify ring 2
set protocols erps ring 1 tc-protection interval 200
set protocols erps ring 1 tc-protection threshold 60

set protocols erps ring 2 control-vlan 20
set protocols erps ring 2 protected-instance 2
set protocols erps ring 2 version 2
set protocols erps ring 2 wtr-timer 6
set protocols erps ring 2 guard-timer 100
set protocols erps ring 2 sub-ring
set protocols erps ring 2 tc-notify ring 1
set protocols erps ring 2 tc-protection interval 200
set protocols erps ring 2 tc-protection threshold 60

set interfaces ge-1/0/1 unit 0 family ethernet-switching vlan members VLAN10
set interfaces ge-1/0/1 unit 0 family ethernet-switching vlan members VLAN100-200
set interfaces ge-1/0/1 unit 0 family ethernet-switching interface-mode trunk
set interfaces ge-1/0/1 unit 0 family ethernet-switching disable false
set protocols erps ring 1 interface ge-1/0/1

set interfaces ge-1/0/2 unit 0 family ethernet-switching vlan members VLAN100-200
set interfaces ge-1/0/2 unit 0 family ethernet-switching vlan members VLAN300-400
set interfaces ge-1/0/2 unit 0 family ethernet-switching interface-mode trunk
set interfaces ge-1/0/2 unit 0 family ethernet-switching disable false
