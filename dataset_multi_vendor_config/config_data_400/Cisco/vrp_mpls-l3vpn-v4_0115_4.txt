mpls ip
!

 no shutdown
 ip address 10.162.1.1 255.255.255.0
 mpls ip
!
interface GigabitEthernet2/0/0
 no shutdown
 ip address 192.168.2.2 255.255.255.0
 mpls ip
!
interface Loopback1
 ip address 10.33.3.9 255.255.255.255
!
interface Loopback2
 ip address 10.33.3.10 255.255.255.255

 neighbor 10.32.2.10 remote-as 100
 neighbor 10.32.2.10 update-source Loopback2
 neighbor 10.32.2.10 ebgp-multihop 3

 neighbor 10.34.4.9 update-source Loopback1
 !

  no synchronization
  neighbor 10.32.2.9 activate
  neighbor 10.34.4.9 activate

 !

  no advertise vpnv4 unicast
  neighbor 10.34.4.9 activate
  neighbor 10.32.2.9 activate

!
