hostname ASBR2
!
mpls label protocol ldp
!
interface GigabitEthernet1/0/0
 no shutdown
 ip address 10.3.1.1 255.255.255.0
 mpls ip
!
interface GigabitEthernet2/0/0
 no shutdown
 ip address 10.2.1.2 255.255.255.0
 mpls ip
!
interface Loopback1
 ip address 3.3.3.3 255.255.255.255
!
router bgp 200
 neighbor 4.4.4.4 remote-as 200
 neighbor 4.4.4.4 update-source Loopback1
 neighbor 10.2.1.1 remote-as 100
 !
 address-family ipv4
  neighbor 4.4.4.4 activate
  neighbor 4.4.4.4 route-map policy2 out
  neighbor 4.4.4.4 send-label
  neighbor 10.2.1.1 activate
  neighbor 10.2.1.1 route-map policy1 out
  neighbor 10.2.1.1 send-label
  network 4.4.4.4 mask 255.255.255.255
  network 10.3.1.0 mask 255.255.255.0
  no synchronization
 exit-address-family
!
router ospf 2
 network 3.3.3.3 0.0.0.0 area 0
 network 10.3.1.0 0.0.0.255 area 0
!
route-map policy1 permit 10
 set mpls label
!
route-map policy2 permit 10
 match mpls label
 set mpls label
!
end
