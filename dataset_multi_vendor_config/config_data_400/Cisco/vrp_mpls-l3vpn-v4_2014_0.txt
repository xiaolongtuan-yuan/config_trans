hostname PE

ip vrf vpna
 rd 100:1
 route-target export 100:100
 route-target import 100:100

interface GigabitEthernet1/0/0
 no shutdown
 ip vrf forwarding vpna
 ip address 10.1.1.1 255.255.255.0

interface GigabitEthernet2/0/0
 no shutdown
 ip vrf forwarding vpna
 ip address 10.2.1.1 255.255.255.0

router bgp 100
 no synchronization
 bgp log-neighbor-changes
 neighbor 10.1.1.2 remote-as 65410
 neighbor 10.1.1.2 update-source GigabitEthernet1/0/0
 neighbor 10.2.1.2 remote-as 65410
 neighbor 10.2.1.2 update-source GigabitEthernet2/0/0

 address-family ipv4
  no synchronization
  network 10.1.1.0 mask 255.255.255.0
  network 10.2.1.0 mask 255.255.255.0
  neighbor 10.1.1.2 activate
  neighbor 10.2.1.2 activate
 exit-address-family

 address-family vpnv4
  neighbor 10.1.1.2 activate
  neighbor 10.2.1.2 activate
  neighbor 10.1.1.2 send-community extended
  neighbor 10.2.1.2 send-community extended
 exit-address-family
