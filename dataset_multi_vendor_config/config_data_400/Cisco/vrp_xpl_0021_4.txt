hostname DeviceE

interface GigabitEthernet3/0/1
 no shutdown
 ip address 1.1.2.1 255.255.255.0

interface GigabitEthernet3/0/2
 no shutdown
 ip address 1.1.1.1 255.255.255.0

router bgp 200
 neighbor 1.1.1.2 remote-as 200
 neighbor 1.1.2.2 remote-as 200
 no synchronization
 address-family ipv4 unicast
  neighbor 1.1.1.2 activate
  neighbor 1.1.1.2 route-map r3 in
  neighbor 1.1.2.2 activate
  neighbor 1.1.2.2 route-map r2 in
 exit-address-family

route-map appmed permit 10
 set metric med

route-map r2 permit 10
 match ip address 10
 set metric 200
 exit

route-map r2 permit 20
 match ip address 20
 exit

route-map r3 permit 10
 match ip address 30
 set metric 100
 exit

route-map r3 permit 20
 exit

access-list 10 permit ip any 2.2.2.0 0.0.0.255
access-list 20 permit ip any 2.2.2.0 0.0.0.255
access-list 20 permit ip any 3.3.3.0 0.0.0.255
access-list 30 permit ip any 2.2.2.0 0.0.0.255

end
