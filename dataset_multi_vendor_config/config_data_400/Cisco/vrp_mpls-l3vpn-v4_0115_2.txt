hostname ASBR1
!
mpls ldp router-id Loopback0
!
interface GigabitEthernet1/0/0
 no shutdown
 ip address 172.16.1.1 255.255.255.0
 mpls ip
 mpls ldp auto-config
!
interface GigabitEthernet2/0/0
 no shutdown
 ip address 192.168.1.1 255.255.255.0
 mpls ip
 mpls ldp auto-config
!
interface Loopback0
 ip address 10.32.2.9 255.255.255.255
!
interface Loopback1
 ip address 10.32.2.10 255.255.255.255
!
router bgp 100
 neighbor 10.31.1.9 remote-as 100
 neighbor 10.31.1.9 update-source Loopback0
 neighbor 10.33.3.10 remote-as 200
 neighbor 10.33.3.10 update-source Loopback1
 neighbor 10.33.3.10 ebgp-multihop 3
 !
 address-family ipv4 unicast
  no synchronization
  neighbor 10.33.3.9 activate
  neighbor 10.31.1.9 activate
 exit-address-family
 !
 address-family vpnv4
  no import vpn-target
  neighbor 10.31.1.9 activate
  neighbor 10.33.3.9 activate
 exit-address-family
!
router ospf 1
 network 10.32.2.9 0.0.0.0 area 0
 network 172.16.1.0 0.0.0.255 area 0
!
router ospf 2
 network 10.32.2.10 0.0.0.0 area 0
 network 192.168.1.0 0.0.0.255 area 0
!
end
