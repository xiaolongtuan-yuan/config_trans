hostname ASBR2
!
mpls label protocol ldp
!
interface GigabitEthernet1/0/0
 no shutdown
 ip address 172.22.1.1 255.255.255.0
 mpls ip
 mpls ldp autoconfig
!
interface GigabitEthernet2/0/0
 no shutdown
 ip address 192.168.1.2 255.255.255.0
 mpls ip
!
interface Loopback1
 ip address 3.3.3.9 255.255.255.255
!
router bgp 200
 neighbor 192.168.1.1 remote-as 100
 neighbor 4.4.4.9 remote-as 200
 neighbor 4.4.4.9 update-source Loopback1
 !
 address-family ipv4
  neighbor 192.168.1.1 activate
  neighbor 4.4.4.9 activate
  no synchronization
 exit-address-family
 !
 address-family l2vpn evpn
  neighbor 4.4.4.9 activate
  neighbor 4.4.4.9 es-evpn-compatible
  neighbor 192.168.1.1 activate
  neighbor 192.168.1.1 es-evpn-compatible
 exit-address-family
!
router ospf 1
 network 3.3.3.9 0.0.0.0 area 0
 network 172.22.1.0 0.0.0.255 area 0
!
end
