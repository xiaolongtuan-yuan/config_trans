sysname DeviceC

multicast routing-enable

ip ip-prefix list-df index 10 permit 192.168.0.0 16 greater-equal 16 less-equal 32

interface GigabitEthernet1/0/0
undo shutdown
ip address 192.168.1.1 255.255.255.0
pim sm
interface GigabitEthernet1/0/0
no shutdown
ip address 192.168.1.1 255.255.255.0
ip pim sparse-mode

interface GigabitEthernet2/0/0
undo shutdown
ip address 192.168.4.1 255.255.255.0
pim sm
interface GigabitEthernet2/0/0
no shutdown
ip address 192.168.4.1 255.255.255.0
ip pim sparse-mode

interface LoopBack0
ip address 1.1.1.1 255.255.255.255
pim sm
interface Loopback0
ip address 1.1.1.1 255.255.255.255
ip pim sparse-mode

pim
c-bsr LoopBack0
c-rp LoopBack0
ip multicast-routing
ip pim rp-address 1.1.1.1
ip pim bsr-candidate Loopback0

ospf 1
area 0.0.0.0
network 192.168.1.0 0.0.0.255
network 192.168.4.0 0.0.0.255
network 1.1.1.1 0.0.0.0
router ospf 1
network 192.168.1.0 0.0.0.255 area 0
network 192.168.4.0 0.0.0.255 area 0
network 1.1.1.1 0.0.0.0 area 0

msdp
peer 192.168.3.2 connect-interface GigabitEthernet 1/0/0
peer 192.168.5.1 connect-interface GigabitEthernet 2/0/0
static-rpf-peer 192.168.3.2 rp-policy list-df
static-rpf-peer 192.168.5.1 rp-policy list-df
ip msdp peer 192.168.3.2 connect-source GigabitEthernet1/0/0
ip msdp peer 192.168.5.1 connect-source GigabitEthernet2/0/0
ip msdp originator-id Loopback0
ip msdp sa-filter in-list list-df
ip msdp sa-filter out-list list-df

return
