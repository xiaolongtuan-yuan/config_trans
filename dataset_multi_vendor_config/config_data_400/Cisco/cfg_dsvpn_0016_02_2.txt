hostname Spoke2

interface GigabitEthernet1/0/0
 ip address 10.1.3.10 255.255.255.0
 tunnel source Loopback0
 tunnel mode gre multipoint
 tunnel destination 172.16.1.1

interface Loopback0
 ip address 192.168.2.1 255.255.255.255

interface Tunnel0
 ip address 172.16.1.3 255.255.255.0
 tunnel source GigabitEthernet1/0/0
 tunnel mode gre multipoint
 tunnel destination 172.16.1.1
 tunnel protection ipsec profile IPSEC_PROFILE

router ospf 1
 router-id 172.16.1.3
 network 172.16.1.0 0.0.0.255 area 0
 network 192.168.2.0 0.0.0.255 area 0

router ospf 2
 router-id 10.1.3.10
 network 10.1.3.0 0.0.0.255 area 1

ip nhrp map multicast dynamic
ip nhrp network-id 1
ip nhrp nhs 172.16.1.1 multicast
ip nhrp shortcut
ip nhrp registration timeout 300
ip nhrp registration holdtime 600

crypto isakmp policy 1
 encr aes
 authentication pre-share
 group 2

crypto isakmp key PRE_SHARED_KEY address 172.16.1.1

crypto ipsec transform-set IPSEC_TRANSFORM esp-aes esp-sha-hmac

crypto ipsec profile IPSEC_PROFILE
 set transform-set IPSEC_TRANSFORM

end
