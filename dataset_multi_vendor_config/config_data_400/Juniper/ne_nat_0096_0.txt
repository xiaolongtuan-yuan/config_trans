set system host-name NATA

set routing-options static route 0.0.0.0/0 next-hop 2.2.2.2

set interfaces ge-1/0/1 unit 0 family inet address 192.168.10.1/24
set interfaces ge-1/0/1 unit 0 family inet nat inside
set interfaces ge-1/0/1 unit 0 filter input route-map-policy1

set interfaces ge-1/0/2 unit 0 family inet address 2.2.2.1/24
set interfaces ge-1/0/2 unit 0 family inet nat outside

set security nat source pool NAT_POOL address 192.168.10.0/24
set security nat source rule-set NAT_RULES from zone trust
set security nat source rule-set NAT_RULES to zone untrust
set security nat source rule-set NAT_RULES rule 1 match source-address 192.168.10.0/24
set security nat source rule-set NAT_RULES rule 1 then source-nat pool NAT_POOL

set firewall family inet filter route-map-policy1 term 10 from source-address 0.0.0.0/0
set firewall family inet filter route-map-policy1 term 10 then routing-instance set-next-hop 1.1.1.100

set protocols ospf area 0 interface ge-1/0/2
set protocols ospf area 0 network 2.2.2.0/24
