set system host-name DeviceA

set vlans VLAN10 vlan-id 10

set interfaces ge-0/0/0 unit 0 family ethernet-switching port-mode trunk
set interfaces ge-0/0/0 unit 0 family ethernet-switching vlan members VLAN10
set interfaces ge-0/0/0 unit 0 family ethernet-switching native-vlan-id 10
set interfaces ge-0/0/0 ethernet-switching-options trust-mode dscp
set interfaces ge-0/0/0 ethernet-switching-options trust-mode cos
set interfaces ge-0/0/0 disable

set interfaces ge-0/0/1 unit 0 family ethernet-switching port-mode trunk
set interfaces ge-0/0/1 unit 0 family ethernet-switching vlan members VLAN10
set interfaces ge-0/0/1 unit 0 family ethernet-switching native-vlan-id 10
set interfaces ge-0/0/1 ethernet-switching-options trust-mode dscp
set interfaces ge-0/0/1 ethernet-switching-options trust-mode cos
set interfaces ge-0/0/1 disable

set class-of-service classifiers dscp EF-CLASS forwarding-class EF
set class-of-service rewrite-rules dscp EF-REWRITE forwarding-class EF loss-priority low code-point 34

set firewall family inet filter QOS-POLICY term EF-CLASS from dscp ef
set firewall family inet filter QOS-POLICY term EF-CLASS then dscp 34
set firewall family inet filter QOS-POLICY term EF-CLASS then accept

set forwarding-options classifiers dscp QOS-POLICY interface ge-0/0/0
set forwarding-options classifiers dscp QOS-POLICY interface ge-0/0/1
