set system host-name CE1

# Enable IPv6 on the interfaces
set interfaces ge-0/0/0 unit 0 family inet6 address 2001:db8:2::2/64
set interfaces lo0 unit 0 family inet6 address 2001:db8:0:1:2::1/128

# Configure BGP
set routing-options router-id 10.10.10.10
set protocols bgp group neighbor-2001:db8:2::1 type external
set protocols bgp group neighbor-2001:db8:2::1 local-address 2001:db8:2::2
set protocols bgp group neighbor-2001:db8:2::1 remote-as 100

# Enable IPv6 address-family for BGP
set protocols bgp group neighbor-2001:db8:2::1 family inet6 unicast
set protocols bgp group neighbor-2001:db8:2::1 family inet6 unicast network 2001:db8:0:1:2::1/128
set protocols bgp group neighbor-2001:db8:2::1 family inet6 unicast peer-as 100
