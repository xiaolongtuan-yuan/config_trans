set system host-name DeviceD

set vlans vlan10 vlan-id 10
set vlans vlan20 vlan-id 20
set vlans vlan100-to-200 vlan-range 100-200
set vlans vlan300-to-400 vlan-range 300-400

set protocols stp region-configuration instance 1 vlan [ 10 100-200 ]
set protocols stp region-configuration instance 2 vlan [ 20 300-400 ]

set protocols erps ring 1 control-vlan 10
set protocols erps ring 1 instance 1
set protocols erps ring 1 wtr-timer 6
set protocols erps ring 1 guard-timer 100

set protocols erps ring 2 control-vlan 20
set protocols erps ring 2 instance 2
set protocols erps ring 2 wtr-timer 6
set protocols erps ring 2 guard-timer 100

set interfaces ge-1/0/1 unit 0 family ethernet-switching interface-mode trunk
set interfaces ge-1/0/1 unit 0 family ethernet-switching vlan members [ vlan10 vlan20 vlan100-to-200 vlan300-to-400 ]
set protocols erps interface ge-1/0/1 ring 1
set protocols erps interface ge-1/0/1 ring 2

set interfaces ge-1/0/2 unit 0 family ethernet-switching interface-mode trunk
set interfaces ge-1/0/2 unit 0 family ethernet-switching vlan members [ vlan10 vlan20 vlan100-to-200 vlan300-to-400 ]
set protocols erps interface ge-1/0/2 ring 1
set protocols erps interface ge-1/0/2 ring 2
