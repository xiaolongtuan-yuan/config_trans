set system host-name ASBR1

# VPN Configuration
set routing-instances vpn1 instance-type virtual-router
set routing-instances vpn1 route-distinguisher 100:2
set routing-instances vpn1 vrf-export 1:1
set routing-instances vpn1 vrf-import 1:1

# MPLS Configuration
set protocols mpls label-protocol ldp
set protocols mpls ldp router-id 2.2.2.9

# Interfaces Configuration
set interfaces ge-0/0/0 unit 0 family inet address 10.10.1.1/24
set interfaces ge-0/0/0 unit 0 family inet mpls

set interfaces ge-0/0/1 unit 0 family inet address 12.12.12.1/24
set interfaces ge-0/0/1 unit 0 family inet vrf vpn1

set interfaces lo0 unit 0 family inet address 2.2.2.9/32

# BGP Configuration
set protocols bgp group external-neighbor type external
set protocols bgp group external-neighbor local-address 2.2.2.9
set protocols bgp group external-neighbor peer-as 100
set protocols bgp group external-neighbor family inet unicast
set protocols bgp group external-neighbor family inet vpnv4
set protocols bgp group external-neighbor send-community both
set protocols bgp group external-neighbor neighbor 1.1.1.9 activate
set protocols bgp group external-neighbor neighbor 12.12.12.2 remote-as 200

# OSPF Configuration
set protocols ospf area 0.0.0.0 interface ge-0/0/0.0
set protocols ospf area 0.0.0.0 interface lo0.0
set protocols ospf area 0.0.0.0 interface ge-0/0/1.0
