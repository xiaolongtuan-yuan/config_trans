set system host-name DeviceC

# Interface Configuration
set interfaces ge-0/0/1 unit 0 family inet address 172.16.4.2/24
set interfaces ge-0/0/3 unit 0 family inet address 172.16.5.1/24

# BGP Configuration
set protocols bgp group 200 neighbor 172.16.4.1 remote-as 65009
set protocols bgp group 200 neighbor 172.16.5.2 remote-as 65009
set protocols bgp group 200 router-id 3.3.3.3

# IPv4 Address Family Configuration
set protocols bgp group 200 address-family ipv4-unicast network 172.16.4.0/24
set protocols bgp group 200 address-family ipv4-unicast network 172.16.5.0/24
set protocols bgp group 200 address-family ipv4-unicast neighbor 172.16.4.1 activate
set protocols bgp group 200 address-family ipv4-unicast neighbor 172.16.5.2 activate
