set system host-name CE

# VLAN Configuration
set vlans vlan10 vlan-id 10
set vlans vlan10 vlan-type private-vlan isolated

# Interface Configuration for Access Ports
set interfaces ge-0/0/1 unit 0 family ethernet-switching vlan members vlan10
set interfaces ge-0/0/2 unit 0 family ethernet-switching vlan members vlan10

# Interface Configuration for Trunk Port
set interfaces ge-0/0/3 unit 0 family ethernet-switching vlan members vlan10
set interfaces ge-0/0/3 unit 0 family ethernet-switching port-mode trunk
set interfaces ge-0/0/3 unit 0 family ethernet-switching trunk allowed vlan 10

# Private VLAN Isolation
set vlans vlan10 private-vlan isolated
