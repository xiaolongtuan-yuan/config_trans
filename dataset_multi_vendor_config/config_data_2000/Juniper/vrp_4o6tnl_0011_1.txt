set system host-name DeviceB

# Interface configuration
set interfaces ge-0/0/0 unit 0 family inet6 address 2001:DB8:100::2/64
set interfaces ge-0/0/1 unit 0 family inet address 172.16.1.2/24
set interfaces lo0 unit 0 family inet6 address 2001:DB8:300::2/64
set interfaces tunnel.1 unit 0 family inet address 192.168.1.2/24
set interfaces tunnel.1 unit 0 tunnel source lo0
set interfaces tunnel.1 unit 0 tunnel destination 2001:DB8:200::1

# OSPF configuration
set protocols ospf area 0 interface ge-0/0/1
set protocols ospf area 0 interface tunnel.1
set protocols ospf area 0 interface lo0

# IPv6 static route
set routing-options static route 2001:DB8:200::/64 next-hop 2001:DB8:100::1
