set system host-name NPE

# Configure IP VRF
set routing-instances vpn1 instance-type virtual-router
set routing-instances vpn1 route-distinguisher 100:1
set routing-instances vpn1 vrf-import 1:1
set routing-instances vpn1 vrf-export 1:1

# Configure MPLS
set protocols mpls label-protocol ldp
set protocols mpls ldp router-id 3.3.3.3
set protocols mpls ip

# Configure ISIS
set protocols isis interface ge-0/0/0.0 address 10.2.1.2/24
set protocols isis interface ge-0/0/0.0 level 1
set protocols isis net 10.0000.0000.0003.00

# Configure Loopback interface
set interfaces lo0 unit 0 family inet address 3.3.3.3/32

# Configure BGP for VPNv6
set protocols bgp group vpn-v6 neighbor 2.2.2.2 remote-as 100
set protocols bgp group vpn-v6 neighbor 2.2.2.2 update-source lo0
set protocols bgp group vpn-v6 family inet6 unicast activate
set protocols bgp group vpn-v6 family inet6 unicast route-map vpn-target in

# Configure IP Prefix List for VPNv6
set policy-options policy-statement vpn-target term 1 from prefix-list default
set policy-options policy-statement vpn-target term 1 then accept

# Apply IPv6 and MPLS on GigabitEthernet interfaces
set interfaces ge-0/0/0 unit 0 family inet address 10.2.1.2/24
set interfaces ge-0/0/0 unit 0 family inet6 address 2001:DB8:30::1/64
set interfaces ge-0/0/0 unit 0 mpls ip

set interfaces ge-0/0/1 unit 0 family inet6 address 2001:DB8:30::1/64
set interfaces ge-0/0/1 unit 0 family inet6 vrf vpn1

# Set route-policy and apply VRF policies for VPNv6
set routing-instances vpn1 protocols bgp group vpnv6 family inet6 unicast policy vpn-target
set routing-instances vpn1 protocols bgp group vpnv6 family inet6 unicast peer 2.2.2.2 enable
