set system host-name DeviceA
set system domain-name huawei.com

# L2TP configuration
set interfaces l2tp-1 unit 0 encapsulation ppp
set protocols l2tp tunnel 1 source-address 10.38.160.1 destination-address 10.38.160.2 password "tunnel-password"
set interfaces l2tp-1 unit 0 ppp chap local-name "username" password "password"

# RADIUS server settings
set system radius-server 10.20.20.1 secret "shared-key"
set system radius-server 10.20.20.1 port 1812 accounting-port 1813
set system aaa authentication-order radius local
set system aaa accounting-order radius

# PPPoE settings
set interfaces ge-0/0/0.100 pppoe-server bind virtual-template 1
set interfaces ge-0/0/0.100 mtu 1492

# Interface settings for GigabitEthernet and Tunnel interfaces
set interfaces ge-0/0/0 unit 0 family inet address 10.38.160.1/24
set interfaces ge-0/0/0.100 unit 0 family inet address 10.38.160.1/24

# Tunnel configuration
set interfaces tunnel.1 unit 0 family inet address unnumbered ge-0/0/0
set interfaces tunnel.1 unit 0 encapsulation l2tp
set interfaces tunnel.1 unit 0 source-address 10.38.160.1
set interfaces tunnel.1 unit 0 destination-address 10.38.160.2

# PPP Authentication
set protocols ppp chap local-name "username"
set protocols ppp chap password "password"

# Enabling AAA
set system aaa authentication order radius local
set system aaa accounting order radius

# VPDN settings
set protocols vpdn group 1 accept-dialin protocol l2tp virtual-template 1
