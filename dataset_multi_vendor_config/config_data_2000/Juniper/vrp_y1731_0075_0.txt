set system host-name CE1

# VLAN configuration
set vlans vlan2 vlan-id 2

# CFM (Connectivity Fault Management) configuration
set protocols cfm version standard
set protocols cfm enable

# Interface configuration for GigabitEthernet1/0/1
set interfaces ge-0/0/1 unit 0 family ethernet-switching interface-mode trunk
set interfaces ge-0/0/1 unit 0 family ethernet-switching vlan members vlan2

# CFM Maintenance Domain and MEP configuration
set protocols cfm maintenance-domain md3
set protocols cfm maintenance-domain md3 ma ma3
set protocols cfm maintenance-domain md3 ma ma3 vlan 2
set protocols cfm maintenance-domain md3 ma ma3 mep 3 interface ge-0/0/1 direction outward
set protocols cfm maintenance-domain md3 ma ma3 mep 3 ccm-send enable
set protocols cfm maintenance-domain md3 ma ma3 remote-mep 4
set protocols cfm maintenance-domain md3 ma ma3 remote-mep 4 ccm-receive enable
set protocols cfm maintenance-domain md3 ma ma3 test 1 mep 3 remote-mep 4
set protocols cfm maintenance-domain md3 ma ma3 loss-measure test-id 1 single-ended-synthetic continual send interval 1000
