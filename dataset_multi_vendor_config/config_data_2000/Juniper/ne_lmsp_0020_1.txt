set system host-name PE2

# MPLS Configuration
set protocols mpls label-protocol ldp
set protocols mpls l2vpn

# Pseudowire Class Configuration
set protocols mpls l2vpn pseudowire-class 3to1 encapsulation mpls

# Interface Configuration
set interfaces ge-0/0/1 unit 0 family inet address 10.2.2.2/24
set interfaces ge-0/0/1 unit 0 family inet mpls
set interfaces lo0 unit 0 family inet address 192.168.1.3/32

# OSPF Configuration
set routing-options router-id 192.168.1.3
set protocols ospf area 0.0.0.0 network 192.168.1.3/32
set protocols ospf area 0.0.0.0 network 10.2.1.0/24

# APS Configuration
set interfaces cpos-3/0/1 unit 0 aps group 1
set interfaces cpos-3/0/2 unit 0 aps group 1 aps protect aps mode one-plus-one unidirectional

# Xconnect Configuration
set interfaces trunk-serial1/1:0 encapsulation mpls
set interfaces trunk-serial1/1:0 mpls l2vc pw-template 3to1 100
set interfaces trunk-serial1/2:0 encapsulation mpls
set interfaces trunk-serial1/2:0 mpls l2vc pw-template 3to1 200

# MPLS LDP Configuration
set protocols mpls ldp remote-peer 192.168.1.2
set protocols mpls ldp remote-ip 192.168.1.2
