set system host-name routerA

# Configure EtherChannel (LAG - Link Aggregation)
set interfaces ae1 unit 0 family inet address 10.40.1.1/24

# Configure Physical Links in the LAG
set interfaces ge-0/0/0 ether-options 802.3ad ae1
set interfaces ge-0/0/1 ether-options 802.3ad ae1

# Configure IP Addresses for the Physical Interfaces
set interfaces ge-0/0/0 unit 0 family inet address 10.1.1.1/24
set interfaces ge-0/0/1 unit 0 family inet address 10.30.1.1/24

# Enable Routing on the Interfaces
set routing-options static route 10.20.1.0/24 next-hop 10.30.1.2 via ge-0/0/1
set routing-options static route 10.20.1.0/24 next-hop 10.40.1.2 via ae1
set routing-options static route 10.50.1.0/24 next-hop 10.30.1.2 via ge-0/0/1
set routing-options static route 10.50.1.0/24 next-hop 10.40.1.2 via ae1

# Enable Load-Balancing for Unequal Cost Multi-Path
set routing-options forwarding-table load-balance unequal-cost
