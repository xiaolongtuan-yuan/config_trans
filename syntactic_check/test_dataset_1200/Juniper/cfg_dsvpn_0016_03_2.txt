set system host-name Spoke2

set interfaces ge-1/0/0 unit 0 family inet address 10.1.3.10/24

set interfaces ge-2/0/0 unit 0 family inet address 192.168.2.1/24

set interfaces gr-1/0/0 unit 0 family inet address 172.16.1.3/24
set interfaces gr-1/0/0 unit 0 tunnel source 10.1.3.10
set interfaces gr-1/0/0 unit 0 tunnel destination 172.16.1.1

set security ike policy IPSEC-PROFILE-NAME mode main
set security ike policy IPSEC-PROFILE-NAME proposal-set basic
set security ike gateway IPSEC-GATEWAY address 172.16.1.1
set security ike gateway IPSEC-GATEWAY ike-policy IPSEC-PROFILE-NAME
set security ipsec policy IPSEC-POLICY proposal-set basic
set security ipsec vpn IPSEC-VPN bind-interface gr-1/0/0.0
set security ipsec vpn IPSEC-VPN ike gateway IPSEC-GATEWAY
set security ipsec vpn IPSEC-VPN ike ipsec-policy IPSEC-POLICY
set security ipsec vpn IPSEC-VPN establish-tunnels immediately

set routing-options router-id 172.16.1.3

set protocols bgp local-as 300
set protocols bgp group external-peers type external
set protocols bgp group external-peers local-address 172.16.1.3
set protocols bgp group external-peers neighbor 172.16.1.1
set protocols bgp group external-peers peer-as 100
set protocols bgp group external-peers family inet unicast

set protocols ospf area 0.0.0.0 interface gr-1/0/0.0
set protocols ospf area 0.0.0.0 interface ge-2/0/0.0

set protocols ospf area 1 interface ge-1/0/0.0
